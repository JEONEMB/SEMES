cmake_minimum_required(VERSION 3.15)
project(ssafy LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(OpenCV 3.4.15 EXACT REQUIRED)
include(CTest)
add_custom_target(check_text_style COMMAND powershell -NoProfile -ExecutionPolicy Bypass -File "${CMAKE_CURRENT_SOURCE_DIR}/tools/check_text_style.ps1")
add_library(alignment STATIC src/alignment.cpp src/common/model_manifest.cpp src/common/sha256.cpp src/common/reference.cpp src/steps/01_find_qr.cpp src/steps/02_find_screws.cpp src/steps/03_compute_yt.cpp src/steps/04_check_result.cpp src/setup/settings.cpp src/setup/find_new_reference.cpp)
target_include_directories(alignment PUBLIC include ${OpenCV_INCLUDE_DIRS} PRIVATE src)
target_link_libraries(alignment PUBLIC ${OpenCV_LIBS})
add_dependencies(alignment check_text_style)
if(MSVC)
    target_compile_options(alignment PRIVATE /utf-8 /EHsc)
    target_compile_definitions(alignment PRIVATE NOMINMAX _CRT_SECURE_NO_WARNINGS)
endif()
function(add_ssafy_program name)
    add_executable(${name} ${ARGN})
    target_link_libraries(${name} PRIVATE alignment)
    target_include_directories(${name} PRIVATE src tools)
    if(MSVC)
        target_compile_options(${name} PRIVATE /utf-8 /EHsc)
        target_compile_definitions(${name} PRIVATE NOMINMAX)
        add_custom_command(TARGET ${name} POST_BUILD COMMAND ${CMAKE_COMMAND} -E copy_if_different "${OpenCV_DIR}/x64/vc15/bin/opencv_world3415$<$<CONFIG:Debug>:d>.dll" "$<TARGET_FILE_DIR:${name}>")
    endif()
endfunction()
add_ssafy_program(measure_one tools/measure_one.cpp)
add_ssafy_program(evaluate_photos tools/evaluate_photos.cpp)
add_ssafy_program(train_combinations tools/train_combinations.cpp tools/common/coefficient_fit.cpp)
if(BUILD_TESTING)
    add_ssafy_program(runtime_tests tools/tests/runtime.cpp)
    target_compile_definitions(runtime_tests PRIVATE SSAFY_MODEL_MANIFEST)
    add_test(NAME runtime_contract COMMAND runtime_tests "${CMAKE_CURRENT_SOURCE_DIR}/data/model")
endif()

if(BUILD_TESTING)
    add_ssafy_program(screw_quality_tests tools/tests/screw_quality.cpp)
    target_include_directories(screw_quality_tests PRIVATE src)
    add_test(NAME screw_quality COMMAND screw_quality_tests "${CMAKE_CURRENT_SOURCE_DIR}/data/model")
endif()
