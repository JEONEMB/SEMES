cmake_minimum_required(VERSION 3.20)
project(lanczos_full_evaluation LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
find_package(OpenCV 3.4.15 REQUIRED PATHS C:/opencv/build)
enable_testing()
add_subdirectory(modules/qrwithbar_lanczos)
add_executable(full_evaluation src/evaluate.cpp)
add_executable(full_train src/full_train.cpp)
target_link_libraries(full_train PRIVATE qrwithbar_lanczos)
target_include_directories(full_train PRIVATE modules/qrwithbar_lanczos/src)
target_compile_options(full_train PRIVATE /utf-8 /EHsc)
target_compile_definitions(full_train PRIVATE NOMINMAX _CRT_SECURE_NO_WARNINGS)
add_custom_command(TARGET full_train POST_BUILD COMMAND ${CMAKE_COMMAND} -E copy_if_different
 "${OpenCV_DIR}/x64/vc15/bin/opencv_world3415.dll" "$<TARGET_FILE_DIR:full_train>")
target_link_libraries(full_evaluation PRIVATE qrwithbar_lanczos)
target_include_directories(full_evaluation PRIVATE modules/qrwithbar_lanczos/src)
target_compile_options(full_evaluation PRIVATE /utf-8 /EHsc)
target_compile_definitions(full_evaluation PRIVATE NOMINMAX _CRT_SECURE_NO_WARNINGS)
add_custom_command(TARGET full_evaluation POST_BUILD COMMAND ${CMAKE_COMMAND} -E copy_if_different
 "${OpenCV_DIR}/x64/vc15/bin/opencv_world3415.dll" "$<TARGET_FILE_DIR:full_evaluation>")
