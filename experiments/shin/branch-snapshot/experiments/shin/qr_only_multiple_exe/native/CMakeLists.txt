cmake_minimum_required(VERSION 3.15)
project(qr_pose_native LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 14)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
find_package(OpenCV REQUIRED COMPONENTS core imgproc imgcodecs video)
add_library(qr_pose STATIC src/pose.cpp src/direct.cpp src/control.cpp src/ssafy_alignment.cpp src/joint_visual_alignment.cpp)
target_include_directories(qr_pose PUBLIC include ${OpenCV_INCLUDE_DIRS})
target_link_libraries(qr_pose PUBLIC ${OpenCV_LIBS})
add_executable(qr_pose_cli src/main.cpp)
target_link_libraries(qr_pose_cli PRIVATE qr_pose)

add_executable(qr_pose_replay src/replay.cpp)
target_link_libraries(qr_pose_replay PRIVATE qr_pose)

set(QR_STAGE_REPO "" CACHE PATH "Original S15P31A507 checkout for the optional Linux device runner")
if(QR_STAGE_REPO)
    if(NOT CMAKE_SYSTEM_NAME STREQUAL "Linux")
        message(FATAL_ERROR "The hardware runner requires Linux and libcamera")
    endif()
    add_subdirectory(${QR_STAGE_REPO}/hardware/stage upstream-stage)
    add_subdirectory(${QR_STAGE_REPO}/hardware/camera upstream-camera)
    add_executable(qr_rig src/rig_device.cpp)
    target_link_libraries(qr_rig PRIVATE qr_pose stage_api stage_camera)
endif()
