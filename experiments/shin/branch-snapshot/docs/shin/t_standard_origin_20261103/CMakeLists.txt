cmake_minimum_required(VERSION 3.20)
project(t_standard_origin LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
find_package(OpenCV REQUIRED)
get_filename_component(REPO "${CMAKE_CURRENT_SOURCE_DIR}/../../qrwithbar-lanczos-src" ABSOLUTE)
set(EXP "${REPO}/experiments/hye/20261102_173432_t_six_methods")
set(BUILD "${REPO}/build-t-only")
foreach(name train combine)
  add_executable(standard_${name} ${name}.cpp)
  target_include_directories(standard_${name} PRIVATE "${EXP}/include" "${REPO}/modules/qrwithbar_lanczos/src" "${REPO}/modules/qrwithbar_lanczos/include" ${OpenCV_INCLUDE_DIRS})
  target_compile_options(standard_${name} PRIVATE /utf-8 /EHsc)
  target_compile_definitions(standard_${name} PRIVATE NOMINMAX)
  target_link_libraries(standard_${name} PRIVATE "${BUILD}/experiments/hye/20261102_173432_t_six_methods/Release/six_methods.lib" "${BUILD}/modules/qrwithbar_lanczos/Release/qrwithbar_lanczos.lib" "${BUILD}/modules/t_estimation/Release/t_model.lib" ${OpenCV_LIBS} bcrypt)
  add_custom_command(TARGET standard_${name} POST_BUILD COMMAND ${CMAKE_COMMAND} -E copy_if_different "${OpenCV_DIR}/x64/vc15/bin/opencv_world3415.dll" "$<TARGET_FILE_DIR:standard_${name}>")
endforeach()
