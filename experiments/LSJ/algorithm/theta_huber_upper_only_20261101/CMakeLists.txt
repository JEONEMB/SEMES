cmake_minimum_required(VERSION 3.15)
project(theta_upper_only LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(OpenCV 3.4 REQUIRED)
set(BASELINE "${CMAKE_CURRENT_SOURCE_DIR}/../theta_huber_20261101")
# Compile only the unchanged point extraction / Huber / Unicode IO modules.
# The baseline two-line pipeline and two-angle decoder are NOT linked.
add_library(upper_alignment_core STATIC
    "${BASELINE}/src/edge_points.cpp" "${BASELINE}/src/huber_line.cpp" "${BASELINE}/src/utf8_io.cpp"
    src/upper_theta.cpp src/alignment.cpp)
target_include_directories(upper_alignment_core PUBLIC include "${BASELINE}/include" ${OpenCV_INCLUDE_DIRS})
target_link_libraries(upper_alignment_core PUBLIC ${OpenCV_LIBS})
add_executable(theta_upper_alignment src/main.cpp)
target_link_libraries(theta_upper_alignment PRIVATE upper_alignment_core)
include(CTest)
if(BUILD_TESTING)
    add_executable(upper_tests tests/upper_tests.cpp)
    target_link_libraries(upper_tests PRIVATE upper_alignment_core)
    add_test(NAME upper_components COMMAND upper_tests)
endif()
foreach(target upper_alignment_core theta_upper_alignment upper_tests)
    if(TARGET ${target} AND MSVC)
        target_compile_options(${target} PRIVATE /utf-8 /W4)
    endif()
endforeach()
if(WIN32)
    foreach(target theta_upper_alignment upper_tests)
        if(TARGET ${target})
            add_custom_command(TARGET ${target} POST_BUILD COMMAND ${CMAKE_COMMAND} -E copy_if_different
                "${OpenCV_DIR}/x64/vc15/bin/opencv_world3415$<$<CONFIG:Debug>:d>.dll" "$<TARGET_FILE_DIR:${target}>")
        endif()
    endforeach()
endif()
