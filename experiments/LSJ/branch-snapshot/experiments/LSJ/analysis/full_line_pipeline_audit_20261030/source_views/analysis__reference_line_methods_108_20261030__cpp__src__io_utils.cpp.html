  21 std::vector<uint8_t> read_bytes(const std::string& utf8_path) {
  22     std::ifstream in(upath(utf8_path), std::ios::binary);
  23     if (!in) throw std::runtime_error("cannot open for reading: " + utf8_path);
  24     return std::vector<uint8_t>(std::istreambuf_iterator<char>(in), std::istreambuf_iterator<char>());
  25 }

  27 void write_bytes(const std::string& utf8_path, const std::vector<uint8_t>& data) {
  28     fs::path p = upath(utf8_path);
  29     if (p.has_parent_path()) fs::create_directories(p.parent_path());
  30     std::ofstream out(p, std::ios::binary);
  31     if (!out) throw std::runtime_error("cannot open for writing: " + utf8_path);
  32     out.write(reinterpret_cast<const char*>(data.data()), static_cast<std::streamsize>(data.size()));
  33 }

  50 std::string join_path(const std::string& a, const std::string& b) {
  51     if (a.empty()) return b;
  52     char last = a.back();
  53     if (last == '/' || last == '\\') return a + b;
  54     return a + "/" + b;

 107 BmpHeader parse_bmp_header(const std::vector<uint8_t>& d) {
 108     BmpHeader h;
 109     if (d.size() < 54 || d[0] != 'B' || d[1] != 'M') return h;
 110     auto i32 = [&](size_t o) { return int32_t(d[o] | (d[o + 1] << 8) | (d[o + 2] << 16) | (d[o + 3] << 24)); };
 111     auto u16 = [&](size_t o) { return int(d[o] | (d[o + 1] << 8)); };
 112     h.width = i32(18);
 113     h.height = i32(22);
 114     if (h.height < 0) h.height = -h.height;
 115     h.bits_per_pixel = u16(28);
 116     h.valid = true;
 117     return h;
 118 }
