  35 def sha(p):
  36     return hashlib.sha256(Path(p).read_bytes()).hexdigest()

  39 def load():
  40     lm = json.loads((RES / "line_measurements.json").read_text(encoding="utf-8"))
  41     targets = json.loads((ROOT / "configs/targets.json").read_text(encoding="utf-8"))["targets"]
  42     manifest = json.loads((ROOT / "configs/inputs_manifest.json").read_text(encoding="utf-8"))
  43     images = {}
  44     for r in manifest["references"]:
  45         raw = np.fromfile(r["path"], dtype=np.uint8)
  46         assert hashlib.sha256(raw.tobytes()).hexdigest() == r["sha256"], r["path"]
  47         images[r["reference_id"]] = cv2.imdecode(raw, cv2.IMREAD_GRAYSCALE)
  48     recs = {r["measurement_id"]: r for r in lm["records"]}
  49     return lm, recs, {t["id"]: t for t in targets}, manifest, images

  56 def show(ax, img, x0, x1, y0, y1, title=None, cmap="gray"):
  57     ax.imshow(img[y0:y1 + 1, x0:x1 + 1], cmap=cmap, vmin=0, vmax=255, extent=(x0 - .5, x1 + .5, y1 + .5, y0 - .5),
  58               interpolation="nearest", aspect="equal")
  59     ax.set_xlim(x0 - .5, x1 + .5); ax.set_ylim(y1 + .5, y0 - .5)

  64 def overlay_mask(ax, rec_images, name, x0, x1, y0, y1, color=(0, 1, 1), alpha=.45):
  65     info = rec_images.get(name)
  66     if not info: return
  67     m = cv2.imdecode(np.fromfile(RES / info["path"], dtype=np.uint8), cv2.IMREAD_GRAYSCALE)
  68     ox, oy = info["origin_xy"]
  69     rgba = np.zeros(m.shape + (4,)); rgba[..., :3] = color; rgba[..., 3] = (m > 0) * alpha
  70     ax.imshow(rgba, extent=(ox - .5, ox + m.shape[1] - .5, oy + m.shape[0] - .5, oy - .5), interpolation="nearest")
  71     ax.set_xlim(x0 - .5, x1 + .5); ax.set_ylim(y1 + .5, y0 - .5)

  74 def draw_line(ax, rec, xa, xb, **kw):
  75     if not rec.get("line") or rec["slope_dy_dx"] is None: return

  77     ax.plot([xa, xb], [m * xa + b, m * xb + b], color=C_FINAL, lw=kw.pop("lw", 1.6), **kw)

  80 def signed_dist(rec, x, y):
  81     (px, py), (dx, dy) = rec["line"]["point_xy"], rec["line"]["direction_xy"]
  82     return -(np.asarray(x) - px) * dy + (np.asarray(y) - py) * dx

  91 def fit_points(it):
  92     pts = np.array(it["intermediates"].get("fit_points", []), dtype=float).reshape(-1, 2)
  93     flag = np.array(it["intermediates"].get("fit_points_inlier_flag", [1] * len(pts)), dtype=bool)
  94     if len(flag) != len(pts): flag = np.ones(len(pts), bool)
  95     return pts, flag

  98 def scatter_points(ax, rec, it, x0, x1, y0, y1):
  99     """Candidates (light), fit inliers (orange), trimmed (purple) inside the view window."""
 100     iv = it["intermediates"]
 101     mid = rec["method_id"][1:]
 102     if mid in ("01", "05", "06") and "columns" in iv:
 103         cols = iv["columns"]
 104         xs = np.array([c[0] for c in cols], float); ys = np.array([c[1] for c in cols], float)
 105         ok = (xs >= x0) & (xs <= x1)
 106         ax.scatter(xs[ok], ys[ok], s=6, marker="s", facecolors="none", edgecolors=C_CAND, lw=.6, label="거친 경계 화소(후보)")
 107     if mid == "02" and "band_points" in iv:
 108         bp = iv["band_points"]
 109         xs = np.array([b[0] for b in bp], float); ys = np.array([b[1] for b in bp], float)
 110         st = np.array([b[6] for b in bp])
 111         ok = (xs >= x0) & (xs <= x1) & (st != "selected")
 112         ax.scatter(xs[ok], ys[ok], s=3, color=C_CAND, label="Devernay 점(제외)")
 113     if mid == "03" and "voting_pixels" in iv:
 114         vp = np.array(iv["voting_pixels"], float).reshape(-1, 3)
 115         ok = (vp[:, 0] >= x0) & (vp[:, 0] <= x1)
 116         ax.scatter(vp[ok & (vp[:, 2] == 0), 0], vp[ok & (vp[:, 2] == 0), 1], s=4, marker="s", color=C_CAND, label="투표 Canny 화소")
 117     pts, flag = fit_points(it)
 118     if len(pts):
 119         ok = (pts[:, 0] >= x0) & (pts[:, 0] <= x1)
 120         ax.scatter(pts[ok & flag, 0], pts[ok & flag, 1], s=4, color=C_SEL, label="선택 지지점(피팅 사용)")
 121         ax.scatter(pts[ok & ~flag, 0], pts[ok & ~flag, 1], s=10, marker="x", color=C_OUT, lw=.8, label="피팅에서 제외(trim)")

 124 def draw_segments(ax, it, x0, x1, y0, y1, lw_sel=2.0):
 125     segs = it["intermediates"].get("segments", [])
 126     for s in segs:
 127         xa, ya, xb, yb, _, _, _, _, st = s
 128         if max(xa, xb) < x0 or min(xa, xb) > x1: continue
 129         if st == "selected": ax.plot([xa, xb], [ya, yb], color=C_SEL, lw=lw_sel, alpha=.9)
 130         elif st.startswith("excluded:not_in_consensus"): ax.plot([xa, xb], [ya, yb], color=C_CAND, lw=1.2)
 131         else: ax.plot([xa, xb], [ya, yb], color=C_EXCL, lw=.8)

 134 def residual_panel(ax, rec, it, t):
 135     pts, flag = fit_points(it)
 136     if rec["status"] != "ok" or not len(pts):
 137         ax.set_title("지지점 잔차: 없음", fontsize=8); fail_text(ax, rec); return
 138     d = signed_dist(rec, pts[:, 0], pts[:, 1])
 139     ax.scatter(pts[flag, 0], d[flag], s=2, color=C_SEL)
 140     ax.scatter(pts[~flag, 0], d[~flag], s=6, marker="x", color=C_OUT, lw=.6)
 141     for a, b in t.get("exclude_x", []): ax.axvspan(a, b, color="gray", alpha=.25, hatch="//")
 142     ax.axhline(0, color=C_FINAL, lw=1)
 143     rms = rec["fit_rms_px"]
 144     ax.set_title(f"최종선 대비 수직 잔차(전 구간) | fit RMS {'-' if rms is None else f'{rms:.3f}'} px", fontsize=8)
 145     ax.set_xlabel("원본 x (px)", fontsize=7); ax.set_ylabel("잔차 (px)", fontsize=7); ax.tick_params(labelsize=6)
 146     lim = max(2.5, float(np.percentile(np.abs(d), 99.5)) * 1.1) if len(d) else 2.5
 147     ax.set_ylim(lim, -lim)

 150 def segment_residual_panel(ax, rec, it, t):
 151     """All detected segments drawn in (x, distance-to-final-line) space."""
 152     if rec["status"] != "ok":
 153         # No final line: draw the detected segments in original (x, y) with the band, y axis stretched.
 154         segs = it["intermediates"].get("segments", [])
 155         y0, y1 = t["band_y"]
 156         for s in segs:
 157             xa, ya, xb, yb, _, _, _, _, st = s
 158             col = C_SEL if st == "selected" else C_CAND if st.startswith("excluded:not_in_consensus") or st.startswith("excluded:occluded") else C_EXCL
 159             ax.plot([xa, xb], [ya, yb], color=col, lw=1.4)
 160         for a, b in t.get("exclude_x", []): ax.axvspan(a, b, color="gray", alpha=.25, hatch="//")
 161         ax.axhline(y0 - .5, color="k", ls=":", lw=.8); ax.axhline(y1 + .5, color="k", ls=":", lw=.8)
 162         ax.set_ylim(y1 + 20, y0 - 20)
 163         ax.set_title(f"검출 선분 {len(segs)}개, 원본 좌표 (세로 확대 표시: 기울기 과장, 각도 판독 금지)\n하늘=가림/짧음 탈락, 회색=각도 등 탈락", fontsize=7)
 164         ax.set_xlabel("원본 x (px)", fontsize=7); ax.tick_params(labelsize=6)
 165         fail_text(ax, rec); return
 166     n = {"selected": 0, "cons": 0, "other": 0}
 167     for s in it["intermediates"].get("segments", []):
 168         xa, ya, xb, yb, _, _, _, _, st = s
 169         da, db = signed_dist(rec, [xa, xb], [ya, yb])
 170         if abs(da) > 40 and abs(db) > 40: continue
 171         if st == "selected": ax.plot([xa, xb], [da, db], color=C_SEL, lw=2.2); n["selected"] += 1
 172         elif st.startswith("excluded:not_in_consensus"): ax.plot([xa, xb], [da, db], color=C_CAND, lw=1.4); n["cons"] += 1
 173         else: ax.plot([xa, xb], [da, db], color=C_EXCL, lw=.9); n["other"] += 1
 174     for a, b in t.get("exclude_x", []): ax.axvspan(a, b, color="gray", alpha=.25, hatch="//")
 175     ax.axhline(0, color=C_FINAL, lw=1)
 176     ax.set_title(f"검출 선분(최종선 대비 거리 공간): 선택 {n['selected']} / 일관성 탈락 {n['cons']} / 필터 탈락 {n['other']}", fontsize=8)
 177     ax.set_xlabel("원본 x (px)", fontsize=7); ax.set_ylabel("최종선까지 거리 (px)", fontsize=7); ax.tick_params(labelsize=6)
 178     ax.set_ylim(20, -20)
