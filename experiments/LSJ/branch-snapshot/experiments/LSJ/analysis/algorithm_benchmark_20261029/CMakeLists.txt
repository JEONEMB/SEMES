cmake_minimum_required(VERSION 3.15)
project(alignment_benchmark LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(BUILD_TESTING OFF CACHE BOOL "Build only the benchmark" FORCE)
if(NOT ALIGNMENT_SOURCE_DIR)
    message(FATAL_ERROR "Set ALIGNMENT_SOURCE_DIR to the frozen payload")
endif()
add_subdirectory("${ALIGNMENT_SOURCE_DIR}" alignment_payload EXCLUDE_FROM_ALL)
add_executable(alignment_benchmark benchmark.cpp)
target_link_libraries(alignment_benchmark PRIVATE alignment_core)
if(MSVC)
    target_compile_options(alignment_benchmark PRIVATE /utf-8 /W4)
endif()
if(WIN32)
    add_custom_command(TARGET alignment_benchmark POST_BUILD COMMAND ${CMAKE_COMMAND} -E copy_if_different
        "${OpenCV_DIR}/x64/vc15/bin/opencv_world3415$<$<CONFIG:Debug>:d>.dll" "$<TARGET_FILE_DIR:alignment_benchmark>")
endif()
