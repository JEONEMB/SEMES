  24 struct Seg {
  25     double x1, y1, x2, y2;
  26     double score = 0;  // detector-specific (LSD -log10 NFA, ELSED none, ...)
  27     std::vector<cv::Point> support;  // integer support pixels (original coordinates)
  28 };
  29 
  30 struct Piece { double xa, xb; };  // visible x interval of a segment
  31 
  32 // Visible part of [xa, xb] inside the target x range minus occlusion intervals.
  33 std::vector<Piece> visible_pieces(const TargetSpec& t, double xa, double xb) {
  34     std::vector<Piece> out;
  35     double a = std::max(xa, t.x0 - 0.5), b = std::min(xb, t.x1 + 0.5);
  36     if (b <= a) return out;
  37     std::vector<Piece> cur = {{a, b}};
  38     for (const auto& iv : t.exclude_x) {
  39         std::vector<Piece> next;
  40         double ea = iv.a - 0.5, eb = iv.b + 0.5;
  41         for (auto pc : cur) {
  42             if (pc.xb <= ea || pc.xa >= eb) { next.push_back(pc); continue; }
  43             if (pc.xa < ea) next.push_back({pc.xa, ea});
  44             if (pc.xb > eb) next.push_back({eb, pc.xb});
  45         }
  46         cur = next;
  47     }
  48     return cur;
  49 }

  51 MethodResult select_and_fit(MethodContext& c, std::vector<Seg> segs, const std::string& detector) {
  52     MethodResult r;
  53     const TargetSpec& t = c.target;
  54     const json& sp = c.common.at("segment_selection").at(t.kind);
  55     const double max_ang = sp.at("max_abs_angle_deg").get<double>();
  56     const double min_len = sp.at("min_visible_len_px").get<double>();
  57     const double off = sp.at("side_offset_px").get<double>();
  58     const double min_con = sp.at("min_contrast").get<double>();
  59     const double tol = sp.at("consensus_inlier_px").get<double>();
  60     const double step = sp.at("sample_step_px").get<double>();
  61     cv::Mat g; c.gray.convertTo(g, CV_64F);  // bilinear sampling on the raw image
  62 
  63     struct Info { Line l; double angle, vis_len = 0, contrast = 0; std::vector<Piece> pieces; std::string state; };
  64     std::vector<Info> info(segs.size());
  65     std::vector<int> cand;
  66     for (size_t i = 0; i < segs.size(); ++i) {
  67         Seg& s = segs[i];
  68         if (s.x2 < s.x1) { std::swap(s.x1, s.x2); std::swap(s.y1, s.y2); }
  69         Info& in = info[i];
  70         in.l = normalize_line(s.x1, s.y1, s.x2 - s.x1, s.y2 - s.y1);
  71         in.angle = line_angle_deg(in.l);
  72         double my = 0.5 * (s.y1 + s.y2);
  73         if (std::fabs(in.angle) > max_ang) { in.state = "excluded:angle"; continue; }
  74         if (!t.in_band(my)) { in.state = "excluded:outside_band"; continue; }
  75         if (std::fabs(s.x2 - s.x1) < 1e-9) { in.state = "excluded:angle"; continue; }
  76         in.pieces = visible_pieces(t, s.x1, s.x2);
  77         for (auto& pc : in.pieces) in.vis_len += (pc.xb - pc.xa) / std::fabs(in.l.dx);
  78         if (in.vis_len < min_len) { in.state = "excluded:occluded_outside_x_or_short"; continue; }
  79         // Polarity: mean intensity on the +normal side (toward +y for dx>0) minus the -normal side.
  80         double sp_sum = 0, sn_sum = 0; int n = 0;
  81         double nx = -in.l.dy, ny = in.l.dx;
  82         for (auto& pc : in.pieces)
  83             for (double x = pc.xa; x <= pc.xb; x += 2.0) {
  84                 double y = s.y1 + (x - s.x1) * in.l.dy / in.l.dx;
  85                 sp_sum += bilinear(g, x + off * nx, y + off * ny);
  86                 sn_sum += bilinear(g, x - off * nx, y - off * ny);
  87                 ++n;
  88             }
  89         in.contrast = n ? (sp_sum - sn_sum) / n : 0;
  90         if (t.polarity * in.contrast < min_con) { in.state = "excluded:wrong_polarity_or_low_contrast"; continue; }
  91         in.state = "candidate";
  92         cand.push_back((int)i);
  93     }
  94     r.n_candidates = (int)segs.size();
  95     r.diagnostics["segments_detected_in_region"] = (int)segs.size();
  96     r.diagnostics["segments_passing_filters"] = (int)cand.size();
  97 
  98     // Consensus line: hypotheses from each candidate and each candidate pair (midpoint to midpoint).
  99     // Score = total visible length of candidates whose visible end points lie within tol of the hypothesis.
 100     auto ends_of = [&](int i, double& xa, double& ya, double& xb, double& yb) {
 101         const Seg& s = segs[i]; const Info& in = info[i];
 102         xa = in.pieces.front().xa; xb = in.pieces.back().xb;
 103         ya = s.y1 + (xa - s.x1) * in.l.dy / in.l.dx; yb = s.y1 + (xb - s.x1) * in.l.dy / in.l.dx;
 104     };
 105     auto score_of = [&](const Line& h, std::vector<int>* members) {
 106         double sc = 0;
 107         for (int i : cand) {
 108             double xa, ya, xb, yb; ends_of(i, xa, ya, xb, yb);
 109             if (std::fabs(signed_distance(h, xa, ya)) <= tol && std::fabs(signed_distance(h, xb, yb)) <= tol) {
 110                 sc += info[i].vis_len;
 111                 if (members) members->push_back(i);
 112             }
 113         }
 114         return sc;
 115     };
 116     std::vector<int> order = cand;
 117     std::sort(order.begin(), order.end(), [&](int a, int b) { return info[a].vis_len > info[b].vis_len; });
 118     if (order.size() > 300) order.resize(300);
 119     double best = -1; Line best_h; std::string best_desc;
 120     for (size_t a = 0; a < order.size(); ++a) {
 121         int i = order[a];
 122         double sc = score_of(info[i].l, nullptr);
 123         if (sc > best) { best = sc; best_h = info[i].l; best_desc = "segment " + std::to_string(i); }
 124         for (size_t b = a + 1; b < order.size(); ++b) {
 125             int j = order[b];
 126             double mxi = 0.5 * (segs[i].x1 + segs[i].x2), myi = 0.5 * (segs[i].y1 + segs[i].y2);
 127             double mxj = 0.5 * (segs[j].x1 + segs[j].x2), myj = 0.5 * (segs[j].y1 + segs[j].y2);
 128             if (std::fabs(mxj - mxi) < 1e-6) continue;
 129             Line h = normalize_line(mxi, myi, mxj - mxi, myj - myi);
 130             if (std::fabs(line_angle_deg(h)) > max_ang) continue;
 131             double s2 = score_of(h, nullptr);
 132             if (s2 > best) { best = s2; best_h = h; best_desc = "pair " + std::to_string(i) + "-" + std::to_string(j); }
 133         }
 134     }
 135     std::vector<int> members;
 136     if (best > 0) score_of(best_h, &members);
 137     for (int i : cand) info[i].state = "excluded:not_in_consensus_line";
 138     for (int i : members) info[i].state = "selected";
 139 
 140     json arr = json::array();
 141     for (size_t i = 0; i < segs.size(); ++i) {
 142         const Seg& s = segs[i]; const Info& in = info[i];
 143         arr.push_back({rnd(s.x1, 4), rnd(s.y1, 4), rnd(s.x2, 4), rnd(s.y2, 4), rnd(in.angle, 5), rnd(in.vis_len, 2),
 144                        rnd(in.contrast, 2), rnd(s.score, 3), in.state});
 145     }
 146     r.intermediates["segments_fields"] = {"x1", "y1", "x2", "y2", "angle_deg", "visible_len_px", "side_contrast", "detector_score", "state"};
 147     r.intermediates["segments"] = arr;
 148     r.diagnostics["consensus_hypothesis"] = best_desc;
 149     r.diagnostics["consensus_score_visible_len_px"] = best;
 150     r.diagnostics["segments_selected"] = (int)members.size();
 151     std::map<std::string, int> state_count;
 152     for (const auto& in : info) state_count[in.state]++;
 153     json sc = json::object();
 154     std::string counts;
 155     for (auto& kv : state_count) { sc[kv.first] = kv.second; counts += " " + kv.first + "=" + std::to_string(kv.second); }
 156     r.diagnostics["segment_state_counts"] = sc;
 157     if (members.empty()) {
 158         r.fail(cand.empty() ? "detection_failed" : "fit_failed",
 159                (cand.empty() ? detector + ": no segment passed angle/band/visibility/polarity filters;" : std::string("no consensus line;")) + counts);
 160         return r;
 161     }
 162     std::vector<cv::Point2d> pts;
 163     for (int i : members) {
 164         const Seg& s = segs[i];
 165         for (auto& pc : info[i].pieces)
 166             for (double x = pc.xa; x <= pc.xb + 1e-9; x += step)
 167                 pts.emplace_back(x, s.y1 + (x - s.x1) * info[i].l.dy / info[i].l.dx);
 168     }
 169     json fitpts = json::array();
 170     for (auto& q : pts) fitpts.push_back(round_pt(q.x, q.y, 4));
 171     r.intermediates["fit_points"] = fitpts;
 172     finish_with_points(r, pts, c.fit,
 173                        "segment_samples: points every 1 px along the visible part of the selected segments (= length weighting); not original edge points");
 174     if (r.has_rms) r.diagnostics["segment_sample_fit_rms_px"] = r.fit_rms;
 175     if (r.has_rms && members.size() == 1) {
 176         // Samples of one segment lie on that segment, so their residual is ~0 by construction.
 177         r.has_rms = false;
 178         r.rms_null_reason = "single selected segment: samples lie on the segment, residual ~0 by construction (see support_pixels_rms_to_final_line_px)";
 179     }
 180     // Diagnostic: the detector's own integer support pixels against the final line.
 181     if (r.has_line) {
 182         std::vector<cv::Point2d> sup;
 183         json supj = json::array();
 184         for (int i : members)
 185             for (auto& q : segs[i].support) {
 186                 if (!t.in_x(q.x) || t.excluded(q.x)) continue;
 187                 sup.emplace_back(q.x, q.y);
 188                 supj.push_back({q.x, q.y});
 189             }
 190         r.intermediates["selected_support_pixels"] = supj;
 191         if (!sup.empty()) {
 192             double s2 = 0;
 193             for (auto& q : sup) { double d = signed_distance(r.line, q.x, q.y); s2 += d * d; }
 194             r.diagnostics["support_pixels_count"] = (int)sup.size();
 195             r.diagnostics["support_pixels_rms_to_final_line_px"] = std::sqrt(s2 / sup.size());
 196             FitResult fr = robust_fit(sup, c.fit);
 197             if (fr.ok) r.diagnostics["support_pixels_refit_angle_deg_diagnostic"] = line_angle_deg(fr.line);
 198             else r.diagnostics["support_pixels_refit_failed"] = fr.reason;
 199         } else {
 200             r.diagnostics["support_pixels_count"] = 0;
 201         }
 202     }
 203     return r;
 204 }

 206 cv::Mat region_u8(MethodContext& c, cv::Rect& reg) {
 207     reg = c.target.detector_region(c.gray.size());
 208     return c.gray(reg).clone();

 213 MethodResult method_lsd(MethodContext& c) {
 214     const json& p = c.params;
 215     cv::Rect reg;
 216     cv::Mat u8 = region_u8(c, reg), f;
 217     u8.convertTo(f, CV_64F);
 218     int n = 0, *reg_img = nullptr, rx = 0, ry = 0;
 219     double* out = LineSegmentDetection(&n, f.ptr<double>(), f.cols, f.rows, p.at("scale").get<double>(),
 220                                        p.at("sigma_scale").get<double>(), p.at("quant").get<double>(),
 221                                        p.at("ang_th").get<double>(), p.at("log_eps").get<double>(),
 222                                        p.at("density_th").get<double>(), p.at("n_bins").get<int>(), &reg_img, &rx, &ry);
 223     std::vector<Seg> segs(n);
 224     for (int i = 0; i < n; ++i) {
 225         // LSD output: origin at the centre of pixel (0,0) (lsd.c adds the 2x2-gradient 0.5 offset).
 226         segs[i] = {out[7 * i] + reg.x, out[7 * i + 1] + reg.y, out[7 * i + 2] + reg.x, out[7 * i + 3] + reg.y, out[7 * i + 6], {}};
 227     }
 228     cv::Mat mask = cv::Mat::zeros(ry, rx, CV_8U);
 229     for (int y = 0; y < ry; ++y)
 230         for (int x = 0; x < rx; ++x) {
 231             int lab = reg_img[x + y * rx];
 232             if (lab > 0 && lab <= n) { segs[lab - 1].support.emplace_back(x + reg.x, y + reg.y); mask.at<uchar>(y, x) = 255; }
 233         }
 234     std::free(out); std::free(reg_img);
 235     MethodResult r = select_and_fit(c, segs, "LSD");
 236     c.sink.save_png("lsd_line_support_regions", mask, reg.tl(), r.images);
 237     r.diagnostics["detector_region_xywh"] = {reg.x, reg.y, reg.width, reg.height};
 238     r.diagnostics["support_pixel_source"] = "LSD reg_img line-support regions (integer pixels)";
 239     return r;
 240 }

 242 MethodResult method_edlines(MethodContext& c) {
 243     const json& p = c.params;
 244     cv::Rect reg;
 245     cv::Mat u8 = region_u8(c, reg);
 246     std::string op = p.at("ed_operator").get<std::string>();
 247     GradientOperator gop = op == "PREWITT" ? PREWITT_OPERATOR : op == "SCHARR" ? SCHARR_OPERATOR : op == "LSD" ? LSD_OPERATOR : SOBEL_OPERATOR;
 248     ED ed(u8, gop, p.at("ed_grad_thresh").get<int>(), p.at("ed_anchor_thresh").get<int>(), p.at("ed_scan_interval").get<int>(),
 249           p.at("ed_min_path_len").get<int>(), p.at("ed_sigma").get<double>(), p.at("ed_sum_flag").get<bool>());
 250     EDLines edl(ed, p.at("line_error").get<double>(), p.at("min_line_len").get<int>(),
 251                 p.at("max_distance_between_two_lines").get<double>(), p.at("max_error").get<double>());
 252     std::vector<LS> lines = edl.getLines();
 253     const std::vector<LineSegment>& info = edl.getLineSegments();
 254     std::vector<std::vector<cv::Point>> chains = edl.getSegments();
 255     std::vector<Seg> segs;
 256     for (size_t i = 0; i < lines.size(); ++i) {
 257         Seg s{lines[i].start.x + reg.x, lines[i].start.y + reg.y, lines[i].end.x + reg.x, lines[i].end.y + reg.y, 0, {}};
 258         if (i < info.size() && info[i].segmentNo >= 0 && info[i].segmentNo < (int)chains.size()) {
 259             const auto& ch = chains[info[i].segmentNo];
 260             int a = std::max(0, info[i].firstPixelIndex), b = std::min((int)ch.size(), info[i].firstPixelIndex + info[i].len);
 261             for (int k = a; k < b; ++k) s.support.emplace_back(ch[k].x + reg.x, ch[k].y + reg.y);
 262             s.score = info[i].len;
 263         }
 264         segs.push_back(s);
 265     }
 266     MethodResult r = select_and_fit(c, segs, "EDLines");
 267     c.sink.save_png("ed_edge_map", edl.getEdgeImage(), reg.tl(), r.images);
 268     r.diagnostics["detector_region_xywh"] = {reg.x, reg.y, reg.width, reg.height};
 269     r.diagnostics["support_pixel_source"] = "ED edge-segment pixels [firstPixelIndex, firstPixelIndex+len) of each EDLines line (collinear joins may not extend this range)";
 270     r.diagnostics["edlines_line_info_count"] = (int)info.size();
 271     return r;
 272 }

 274 MethodResult method_fld(MethodContext& c) {
 275     const json& p = c.params;
 276     cv::Rect reg;
 277     cv::Mat u8 = region_u8(c, reg);
 278     auto fld = lsj_fld::createFastLineDetectorImpl(p.at("length_threshold").get<int>(), p.at("distance_threshold").get<float>(),
 279                                                    p.at("canny_th1").get<double>(), p.at("canny_th2").get<double>(),
 280                                                    p.at("canny_aperture_size").get<int>(), p.at("do_merge").get<bool>());
 281     std::vector<cv::Vec4f> lines;
 282     std::vector<std::vector<cv::Point2i>> support;
 283     lsj_fld::detectWithSupport(fld, u8, lines, support);
 284     std::vector<Seg> segs;
 285     for (size_t i = 0; i < lines.size(); ++i) {
 286         Seg s{lines[i][0] + (double)reg.x, lines[i][1] + (double)reg.y, lines[i][2] + (double)reg.x, lines[i][3] + (double)reg.y, 0, {}};
 287         if (i < support.size()) for (auto& q : support[i]) s.support.emplace_back(q.x + reg.x, q.y + reg.y);
 288         segs.push_back(s);
 289     }
 290     MethodResult r = select_and_fit(c, segs, "FLD");
 291     cv::Mat canny;  // same Canny call FLD makes internally (for display only)
 292     cv::Canny(u8, canny, p.at("canny_th1").get<double>(), p.at("canny_th2").get<double>(), p.at("canny_aperture_size").get<int>());
 293     c.sink.save_png("fld_internal_canny_recomputed", canny, reg.tl(), r.images);
 294     r.diagnostics["detector_region_xywh"] = {reg.x, reg.y, reg.width, reg.height};
 295     r.diagnostics["support_pixel_source"] = "FLD chain pixels (l_points) used in each segment's fitLine";
 296     r.diagnostics["support_list_aligned"] = support.size() == lines.size();
 297     return r;
 298 }

 300 MethodResult method_elsed(MethodContext& c) {
 301     const json& p = c.params;
 302     cv::Rect reg;
 303     cv::Mat u8 = region_u8(c, reg);
 304     upm::ELSEDParams ep;
 305     ep.ksize = p.at("ksize").get<int>();
 306     ep.sigma = p.at("sigma").get<float>();
 307     ep.gradientThreshold = p.at("gradientThreshold").get<float>();
 308     ep.anchorThreshold = (uint8_t)p.at("anchorThreshold").get<int>();
 309     ep.scanIntervals = p.at("scanIntervals").get<unsigned int>();
 310     ep.minLineLen = p.at("minLineLen").get<int>();
 311     ep.lineFitErrThreshold = p.at("lineFitErrThreshold").get<double>();
 312     ep.pxToSegmentDistTh = p.at("pxToSegmentDistTh").get<double>();
 313     ep.junctionEigenvalsTh = p.at("junctionEigenvalsTh").get<double>();
 314     ep.junctionAngleTh = p.at("junctionAngleTh_deg").get<double>() * (3.14159265358979323846 / 180.0);
 315     ep.validationTh = p.at("validationTh").get<double>();
 316     ep.validate = p.at("validate").get<bool>();
 317     ep.treatJunctions = p.at("treatJunctions").get<bool>();
 318     upm::ELSED elsed(ep);
 319     upm::Segments out = elsed.detect(u8);
 320     const upm::ImageEdges& pix = elsed.getValidSegmentPixels();
 321     std::vector<Seg> segs;
 322     for (size_t i = 0; i < out.size(); ++i) {
 323         Seg s{out[i][0] + (double)reg.x, out[i][1] + (double)reg.y, out[i][2] + (double)reg.x, out[i][3] + (double)reg.y, 0, {}};
 324         if (i < pix.size()) for (auto& q : pix[i]) s.support.emplace_back(q.x + reg.x, q.y + reg.y);
 325         segs.push_back(s);
 326     }
 327     MethodResult r = select_and_fit(c, segs, "ELSED");
 328     r.diagnostics["detector_region_xywh"] = {reg.x, reg.y, reg.width, reg.height};
 329     r.diagnostics["support_pixel_source"] = "ELSED FullSegmentInfo pixels of each returned segment";
 330     r.diagnostics["support_list_aligned"] = pix.size() == out.size();
 331     return r;
 332 }
