  12 OUT = Path(__file__).resolve().parent

  21 source_files: dict[str, str] = {}

  25 def sha(path: Path) -> str:
  26     return hashlib.sha256(path.read_bytes()).hexdigest()
  27 
  28 
  29 def read_source(relative: str):
  30     path = SOURCE / relative
  31     source_files[relative] = sha(path)
  32     return json.loads(path.read_text(encoding='utf-8-sig'))
  33

  35 def write_json(relative: str, data):
  36     path = OUT / relative
  37     path.parent.mkdir(parents=True, exist_ok=True)
  38     with path.open('w', encoding='utf-8') as stream:
  39         json.dump(data, stream, ensure_ascii=False, indent=2, allow_nan=False)
  40         stream.write('\n')

  43 def copy_source(relative: str, destination: str | None = None):
  44     src, dst = SOURCE / relative, OUT / (destination or relative)
  45     digest = sha(src)
  46     source_files[relative] = digest
  47     if dst.exists() and sha(dst) != digest:
  48         raise ValueError(f'Existing copy differs from source: {dst}')
  49     dst.parent.mkdir(parents=True, exist_ok=True)
  50     if not dst.exists():
  51         shutil.copyfile(src, dst)
  52     if sha(dst) != digest:
  53         raise ValueError(f'Copy hash mismatch: {dst}')
  54     copied_files.append({'source': relative, 'destination': str(dst.relative_to(OUT)),
  55                          'sha256': digest})

  58 def main():
  59     source_files.clear()
  60     copied_files.clear()
  61     measurements = read_source('results/line_measurements.json')
  62     original_combinations = read_source('results/combinations.json')
  63     original_methods = read_source('results/method_status.json')
  64     rows = measurements['records']
  65     assert len({r['measurement_id'] for r in rows}) == len(rows)
  66     rankings, selected = {}, {}
  67     for family, targets in TARGETS.items():
  68         prefix = 'E' if family == 'external' else 'Q'
  69         methods = []
  70         for method_id in sorted({r['method_id'] for r in rows if r['method_id'].startswith(prefix)}):
  71             rr = [r for r in rows if r['method_id'] == method_id]
  72             expected = {(ref, target) for ref in REFERENCES for target in targets}
  73             actual = {(r['reference_id'], r['target_id']) for r in rr}
  74             failed = [r['measurement_id'] for r in rr
  75                       if r['status'] != 'ok' or not isinstance(r.get('image_angle_deg'), (int, float))
  76                       or not math.isfinite(r['image_angle_deg'])]
  77             complete = actual == expected and len(rr) == len(expected) and not failed
  78             item = {'method_id': method_id, 'method_name': rr[0]['method_name'],
  79                     'eligible_all_required_slots_ok': complete,
  80                     'failed_measurement_ids': failed,
  81                     'missing_slots': sorted(expected - actual), 'targets': {}}
  82             for target in targets:
  83                 tr = [r for r in rr if r['target_id'] == target and r['status'] == 'ok']
  84                 angles = {r['reference_id']: r['image_angle_deg'] for r in tr}
  85                 target_complete = len(angles) == 4 and all(isinstance(v, (int, float)) and math.isfinite(v) for v in angles.values())
  86                 item['targets'][target] = {
  87                     'image_angles_deg': angles,
  88                     'range_deg': max(angles.values()) - min(angles.values()) if target_complete else None}
  89             item['score_deg'] = max(t['range_deg'] for t in item['targets'].values()) if complete else None
  90             methods.append(item)
  91         methods.sort(key=lambda m: (m['score_deg'] if m['score_deg'] is not None else math.inf, m['method_id']))
  92         eligible = [m for m in methods if m['eligible_all_required_slots_ok']]
  93         selected[family] = ([m['method_id'] for m in eligible[:KEEP_PER_FAMILY[family]]]
  94                             if family == 'external' else QR_METHODS.copy())
  95         assert set(selected[family]) <= {m['method_id'] for m in eligible}
  96         for item in methods:
  97             item['rank_among_complete_methods'] = eligible.index(item) + 1 if item in eligible else None
  98             item['selected'] = item['method_id'] in selected[family]
  99             item['reason'] = ('상위 3개 후보로 선별' if item['selected'] else
 100                               '추출 누락 또는 실패가 있어 이번 선별에서 제외' if not item['eligible_all_required_slots_ok'] else
 101                               '완전 추출되었으나 이번 상위 3개 선별에서 보류; 사용 불가 판정 아님')
 102             if family == 'qr':
 103                 item['reason'] = ('사용자 지정: Q02·Q04만 사용' if item['selected'] else
 104                                   '사용자 지정으로 제외: 초기 3위 후보였으나 Q02·Q04만 사용' if item['method_id'] == 'Q01' else
 105                                   '보류: 사용자 지정 Q02·Q04에 포함되지 않음; 사용 불가 판정 아님')
 106         rankings[family] = methods

 114     assert len(selected_combinations) == 24 and all(c['status'] == 'complete' for c in selected_combinations)
 115     assert all(set(c['measurement_refs'].values()) <= measurement_ids for c in selected_combinations)

 118     write_json('results/line_measurements.json', {'meta': metadata, 'records': selected_rows})
 119     for family in TARGETS:
 120         subset = [r for r in selected_rows if r['method_id'] in selected[family]]
 121         write_json(f'results/{family}_lines.json', {'meta': metadata, 'records': subset})

 123                'meaning': original_combinations['meta']['meaning']}, 'combinations': selected_combinations})
 124     write_json('results/method_status.json', {'meta': metadata,
 125                'methods': [m for m in original_methods['methods'] if m['method_id'] in selected_ids]})
 126     write_json('rankings.json', {'ranking_measure': 'max over targets(max over references(angle) - min over references(angle))',
 127                                'units': 'image angle degree; not robot theta', 'rankings': rankings})

 133     for method_id in selected_ids:

 136     for name in ['inputs_manifest.json', 'targets.json', 'methods.json', 'method_sources.json']:
