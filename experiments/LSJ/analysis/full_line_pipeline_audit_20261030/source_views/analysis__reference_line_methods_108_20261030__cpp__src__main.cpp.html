  27 std::vector<std::string> split(const std::string& s) {
  28     std::vector<std::string> v;
  29     std::stringstream ss(s);
  30     std::string item;
  31     while (std::getline(ss, item, ',')) if (!item.empty()) v.push_back(item);
  32     return v;
  33 }

  35 // Config keys are a single id ("Q10") or a shared family ("E01/Q01").
  36 const json& method_config(const json& methods, const std::string& id) {
  37     for (auto it = methods.begin(); it != methods.end(); ++it) {
  38         std::string key = it.key();
  39         size_t slash = key.find('/');
  40         if (key == id || (slash != std::string::npos && (key.substr(0, slash) == id || key.substr(slash + 1) == id))) return it.value();
  41     }
  42     throw std::runtime_error("no config for method " + id);
  43 }

  47            u8"바깥 밝은 테두리 상단 경계(Q01~Q11)와 다른 특징이다.";

  50 TargetSpec make_target(const json& j, const json& common) {
  51     TargetSpec t;
  52     t.id = j.at("id").get<std::string>();
  53     t.kind = j.at("kind").get<std::string>();
  54     t.feature_definition = j.at("feature_definition").get<std::string>();
  55     t.x0 = j.at("x_range")[0].get<int>(); t.x1 = j.at("x_range")[1].get<int>();
  56     t.y0 = j.at("band_y")[0].get<int>(); t.y1 = j.at("band_y")[1].get<int>();
  57     for (auto& e : j.at("exclude_x")) t.exclude_x.push_back({e[0].get<int>(), e[1].get<int>()});
  58     t.polarity = j.at("polarity").get<int>();
  59     t.margin = common.at("detector_margin_px").at(t.kind).get<int>();
  60     t.column_step = common.at("column_step_px").at(t.kind).get<int>();
  61     if (j.contains("roi_xyxy")) {
  62         auto r = j.at("roi_xyxy");
  63         t.roi = cv::Rect(r[0].get<int>(), r[1].get<int>(), r[2].get<int>() - r[0].get<int>(), r[3].get<int>() - r[1].get<int>());
  64     } else {
  65         t.roi = cv::Rect(t.x0, t.y0, t.x1 - t.x0 + 1, t.y1 - t.y0 + 1);
  66     }
  67     t.raw = j;
  68     return t;
  69 }

  71 json roi_json(const TargetSpec& t) {
  72     json j;
  73     j["x_range_inclusive"] = {t.x0, t.x1};
  74     j["band_y_inclusive"] = {t.y0, t.y1};
  75     json ex = json::array();
  76     for (auto& iv : t.exclude_x) ex.push_back({iv.a, iv.b});
  77     j["exclude_x_inclusive"] = ex;
  78     if (t.kind == "qr") j["roi_xyxy_exclusive_end"] = {t.roi.x, t.roi.y, t.roi.x + t.roi.width, t.roi.y + t.roi.height};
  79     j["detector_margin_px"] = t.margin;
  80     j["column_step_px"] = t.column_step;
  81     j["polarity"] = t.polarity;
  82     return j;
  83 }

  85 std::string csv_escape(const std::string& s) {
  86     std::string o = "\"";
  87     for (char c : s) { if (c == '"') o += "\"\""; else o += c; }
  88     return o + "\"";
  89 }

  91 std::string num_or_empty(const json& v) {
  92     if (v.is_null()) return "";
  93     std::ostringstream os;
  94     os.precision(17);
  95     os << v.get<double>();
  96     return os.str();
  97 }

 110     std::string root = ".", out;
 111     std::set<std::string> want_refs, want_methods;
 112     for (size_t i = 0; i < args.size(); ++i) {
 113         if (args[i] == "--root" && i + 1 < args.size()) root = args[++i];
 114         else if (args[i] == "--out" && i + 1 < args.size()) out = args[++i];
 115         else if (args[i] == "--refs" && i + 1 < args.size()) for (auto& s : split(args[++i])) want_refs.insert(s);
 116         else if (args[i] == "--methods" && i + 1 < args.size()) for (auto& s : split(args[++i])) want_methods.insert(s);
 117         else { std::cerr << "unknown argument: " << args[i] << "\n"; return 2; }
 118     }
 119     if (out.empty()) out = join_path(root, "results");
 120     try {
 121         cv::setNumThreads(1);
 122         auto t_start = std::chrono::steady_clock::now();
 123         const std::string cfg_dir = join_path(root, "configs");
 124         std::map<std::string, std::string> cfg_hash;
 125         auto load = [&](const std::string& name) {
 126             std::string path = join_path(cfg_dir, name);
 127             auto bytes = read_bytes(path);
 128             cfg_hash[name] = sha256_hex(bytes);
 129             return json::parse(std::string(bytes.begin(), bytes.end()));
 130         };
 131         json manifest = load("inputs_manifest.json");
 132         json targets_cfg = load("targets.json");
 133         json methods_cfg = load("methods.json");

 141         for (auto& m : registry())
 142             if (want_methods.empty() || want_methods.count(m.id)) methods.push_back(&m);

 145         json records = json::array();
 146         json inputs = json::array();
 147         std::map<std::string, std::string> status_of;  // measurement_id -> status
 148         std::map<std::string, std::map<std::string, int>> method_counts;  // method -> status -> count
 149 
 150         for (auto& ref : manifest.at("references")) {
 151             const std::string rid = ref.at("reference_id").get<std::string>();
 152             if (!want_refs.empty() && !want_refs.count(rid)) continue;
 153             const std::string path = ref.at("path").get<std::string>();
 154             json in = {{"reference_id", rid}, {"path", path}, {"expected_sha256", ref.at("sha256")}};
 155             std::string problem;
 156             cv::Mat gray;
 157             std::string digest;
 158             try {
 159                 auto bytes = read_bytes(path);
 160                 digest = sha256_hex(bytes);
 161                 BmpHeader h = parse_bmp_header(bytes);
 162                 in["sha256_before"] = digest;
 163                 in["bmp_header"] = {{"valid", h.valid}, {"width", h.width}, {"height", h.height}, {"bits_per_pixel", h.bits_per_pixel}};
 164                 if (digest != ref.at("sha256").get<std::string>()) problem = "sha256 mismatch";
 165                 else if (!h.valid || h.width != manifest.at("expected_width").get<int>() || h.height != manifest.at("expected_height").get<int>() ||
 166                          h.bits_per_pixel != manifest.at("expected_bits_per_pixel").get<int>()) problem = "bmp header mismatch";
 167                 else {
 168                     gray = decode_gray(bytes);
 169                     if (gray.empty() || gray.cols != h.width || gray.rows != h.height) problem = "decode failed or size mismatch";
 170                 }
 171             } catch (const std::exception& e) {
 172                 problem = std::string("read failed: ") + e.what();
 173             }
 174             in["status"] = problem.empty() ? "ok" : "mismatch";
 175             if (!problem.empty()) in["problem"] = problem;
 176             std::cout << rid << " input " << (problem.empty() ? "ok" : problem) << std::endl;
 177 
 178             for (const MethodInfo* m : methods) {
 179                 const std::string kind = m->family == "E" ? "external" : "qr";
 180                 const json& mcfg = method_config(methods_cfg.at("methods"), m->id);
 181                 for (const TargetSpec& t : targets) {
 182                     if (t.kind != kind) continue;
 183                     const std::string mid = rid + "_" + m->id + "_" + t.id;
 184                     json params = mcfg.at(kind);
 185                     json hashed = {{"method", params}, {"common_fit", common.at("fit")}, {"segment_selection", common.at("segment_selection").at(kind)},
 186                                    {"roi", roi_json(t)}};
 187                     std::string dumped = hashed.dump();
 188                     std::string config_hash = sha256_hex(std::vector<uint8_t>(dumped.begin(), dumped.end()));
 189                     std::string rel_dir = "intermediates/" + rid + "/" + m->id;
 190                     Sink sink{join_path(out, rel_dir), rel_dir, t.id};
 191                     make_dirs(sink.abs_dir);
 192                     MethodResult res;
 193                     auto t0 = std::chrono::steady_clock::now();
 194                     if (!problem.empty()) {
 195                         res.fail("input_mismatch", problem);
 196                     } else {
 197                         MethodContext ctx{gray, t, params, common, fit_params_for(common, kind), sink};
 198                         try {
 199                             res = m->fn(ctx);
 200                         } catch (const std::exception& e) {
 201                             res = MethodResult();
 202                             res.fail("exception", e.what());
 203                         }
 204                     }
 205                     double ms = std::chrono::duration<double, std::milli>(std::chrono::steady_clock::now() - t0).count();
 206                     json rec;
 207                     rec["measurement_id"] = mid;
 208                     rec["reference_id"] = rid;
 209                     rec["dataset"] = ref.at("dataset");
 210                     rec["input_path"] = path;
 211                     rec["input_sha256"] = digest;
 212                     rec["method_id"] = m->id;
 213                     rec["method_name"] = m->name_ko;
 214                     rec["implementation"] = m->implementation;
 215                     rec["target_id"] = t.id;
 216                     rec["target_kind"] = kind;
 217                     rec["feature_definition"] = m->id == "Q12" ? q12_definition(t.id.substr(3)) : t.feature_definition;
 218                     rec["feature_class"] = m->id == "Q12" ? "qr_finder_center_baseline" : (kind == "qr" ? "qr_outer_border_top_edge" : "external_structure_edge");
 219                     rec["status"] = res.status;
 220                     rec["failure_category"] = res.status == "ok" ? json(nullptr) : json(res.failure_category);
 221                     rec["failure_reason"] = res.status == "ok" ? json(nullptr) : json(res.failure_reason);
 222                     rec["roi"] = roi_json(t);
 223                     rec["parameters"] = params;
 224                     rec["config_hash"] = config_hash;
 225                     if (res.has_line) {
 226                         json lj = line_json(res.line);
 227                         rec["line"] = lj;
 228                         rec["slope_dy_dx"] = lj["slope_dy_dx"];
 229                         rec["intercept_y_px"] = lj["intercept_y_px"];
 230                         rec["image_angle_deg"] = lj["image_angle_deg"];
 231                         rec["line_endpoints_xy"] = res.has_endpoints ? json::array({json::array({res.ex1, res.ey1}), json::array({res.ex2, res.ey2})}) : json(nullptr);
 232                         rec["draw_endpoints_at_x_range"] = line_draw_endpoints(res.line, t.x0, t.x1);
 233                     } else {
 234                         rec["line"] = nullptr;
 235                         rec["slope_dy_dx"] = nullptr;
 236                         rec["intercept_y_px"] = nullptr;
 237                         rec["image_angle_deg"] = nullptr;
 238                         rec["line_endpoints_xy"] = nullptr;
 239                         rec["draw_endpoints_at_x_range"] = nullptr;
 240                     }
 241                     rec["n_candidates"] = res.n_candidates;
 242                     rec["n_selected"] = res.n_selected;
 243                     rec["support_type"] = res.support_type.empty() ? json(nullptr) : json(res.support_type);
 244                     if (res.has_rms) { rec["fit_rms_px"] = res.fit_rms; rec["fit_rms_null_reason"] = nullptr; }
 245                     else {
 246                         rec["fit_rms_px"] = nullptr;
 247                         rec["fit_rms_null_reason"] = !res.rms_null_reason.empty() ? res.rms_null_reason : std::string("no line");
 248                     }
 249                     rec["fit_rms_note"] = "fit RMS is the residual of the support points to the fitted line, not measurement accuracy";
 250                     rec["diagnostics"] = res.diagnostics;
 251                     std::string inter_rel = rel_dir + "/" + t.id + ".json";
 252                     rec["intermediate_json"] = inter_rel;
 253                     rec["intermediate_images"] = res.images;
 254                     rec["runtime_ms"] = ms;
 255                     json inter = {{"measurement_id", mid}, {"intermediates", res.intermediates}, {"images", res.images}};
 256                     write_text(join_path(out, inter_rel), inter.dump(-1, ' ', false, json::error_handler_t::replace));
 257                     records.push_back(rec);
 258                     status_of[mid] = res.status;
 259                     method_counts[m->id][res.status]++;
 260                     std::printf("%s %s %.1f ms%s%s\n", mid.c_str(), res.status.c_str(), ms,
 261                                 res.status == "ok" ? "" : " : ", res.status == "ok" ? "" : res.failure_reason.c_str());
 262                     std::fflush(stdout);
 263                 }
 264             }
 265             if (!digest.empty()) {
 266                 std::string after = sha256_hex(read_bytes(path));
 267                 in["sha256_after"] = after;
 268                 in["unchanged_after_run"] = after == digest;
 269             }
 270             inputs.push_back(in);
 271         }
