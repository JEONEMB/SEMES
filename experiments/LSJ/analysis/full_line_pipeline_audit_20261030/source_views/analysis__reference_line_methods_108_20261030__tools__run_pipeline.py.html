  32 def sha(path):
  33     return hashlib.sha256(Path(path).read_bytes()).hexdigest()

  36 def run(step, cmd, cwd=ROOT):
  37     start = time.time()
  38     proc = subprocess.run(cmd, cwd=cwd, capture_output=True)
  39     elapsed = time.time() - start
  40     out = proc.stdout.decode("utf-8", errors="replace")
  41     err = proc.stderr.decode("utf-8", errors="replace")
  42     (LOGS / f"{step}.stdout.txt").write_text(out, encoding="utf-8")
  43     (LOGS / f"{step}.stderr.txt").write_text(err, encoding="utf-8")
  44     rec = {"step": step, "command": cmd, "cwd": str(cwd), "exit_code": proc.returncode, "wall_time_s": round(elapsed, 3),
  45            "started": datetime.fromtimestamp(start).isoformat(timespec="seconds")}
  46     (LOGS / f"{step}.json").write_text(json.dumps(rec, ensure_ascii=False, indent=2), encoding="utf-8")
  47     print(f"[{step}] exit={proc.returncode} {elapsed:.1f}s", flush=True)
  48     return proc.returncode, out

  51 def prepare_results(path):
  52     """Clear a results folder only if this pipeline created it; never touch other files."""
  53     if path.exists():
  54         if not (path / MARKER).exists() and any(path.iterdir()):
  55             raise SystemExit(f"{path} has files not created by this pipeline; refusing to overwrite")
  56         shutil.rmtree(path)
  57     path.mkdir(parents=True)
  58     (path / MARKER).write_text("created by tools/run_pipeline.py\n", encoding="utf-8")

  61 def numeric_fields(rec):
  62     keys = ["status", "slope_dy_dx", "intercept_y_px", "image_angle_deg", "fit_rms_px", "n_candidates", "n_selected"]
  63     return {k: rec.get(k) for k in keys}

  66 def compare_runs(a_path, b_path):
  67     a = {r["measurement_id"]: numeric_fields(r) for r in json.loads(Path(a_path).read_text(encoding="utf-8"))["records"]}
  68     b = {r["measurement_id"]: numeric_fields(r) for r in json.loads(Path(b_path).read_text(encoding="utf-8"))["records"]}
  69     diffs = []
  70     for mid in sorted(set(a) & set(b)):
  71         for k, va in a[mid].items():
  72             vb = b[mid][k]
  73             if isinstance(va, float) and isinstance(vb, float):
  74                 if va != vb:
  75                     diffs.append({"measurement_id": mid, "field": k, "a": va, "b": vb, "abs_diff": abs(va - vb)})
  76             elif va != vb:
  77                 diffs.append({"measurement_id": mid, "field": k, "a": va, "b": vb})
  78     return {"compared_ids": len(set(a) & set(b)), "only_in_a": sorted(set(a) - set(b)), "only_in_b": sorted(set(b) - set(a)),
  79             "differences": diffs, "identical": not diffs}

  82 def tree_hashes(base, patterns):
  83     out = {}
  84     for pat in patterns:
  85         for p in sorted(base.glob(pat)):
  86             if p.is_file():
  87                 out[str(p.relative_to(ROOT)).replace("\\", "/")] = sha(p)
  88     return out
