  16 struct Interval {
  17     int a = 0, b = 0;  // inclusive
  18 };

  20 struct TargetSpec {
  21     std::string id;    // upper_structure, lower_structure, qr_L, qr_C, qr_R
  22     std::string kind;  // "external" | "qr"
  23     std::string feature_definition;
  24     int x0 = 0, x1 = 0;  // support x range, inclusive
  25     int y0 = 0, y1 = 0;  // search band, inclusive
  26     std::vector<Interval> exclude_x;  // inclusive
  27     int polarity = 1;  // +1: intensity increases with y across the target boundary
  28     cv::Rect roi;      // QR: roi_xyxy (exclusive end). External: band x/y range (no margin).
  29     int margin = 0;    // detector context margin, px
  30     int column_step = 1;
  31     json raw;
  32 
  33     bool excluded(double x) const {
  34         for (const auto& iv : exclude_x)
  35             if (x >= iv.a - 0.5 && x <= iv.b + 0.5) return true;
  36         return false;
  37     }
  38     bool in_x(double x) const { return x >= x0 - 0.5 && x <= x1 + 0.5; }
  39     bool in_band(double y) const { return y >= y0 - 0.5 && y <= y1 + 0.5; }
  40     // Detector region: target area plus context margin, clamped to the image.
  41     cv::Rect detector_region(const cv::Size& image) const;
  42 };

  44 // Final line in original coordinates.
  45 struct Line {
  46     double px = 0, py = 0;  // reference point
  47     double dx = 1, dy = 0;  // unit direction, dx > 0 (or dx == 0 and dy < 0)
  48 };

  53 json line_json(const Line& l);                    // slope/intercept/general form/angle
  54 json line_draw_endpoints(const Line& l, double xa, double xb);

  79 // Output of one method on one target in one image.

  82     std::string failure_category;        // detection_failed | boundary_confusion | corner_not_identified | not_implemented | fit_failed
  83     std::string failure_reason;
  84     bool has_line = false;
  85     Line line;
  86     bool has_endpoints = false;
  87     double ex1 = 0, ey1 = 0, ex2 = 0, ey2 = 0;
  88     int n_candidates = 0;
  89     int n_selected = 0;
  90     std::string support_type;
  91     bool has_rms = false;
  92     double fit_rms = 0;
  93     std::string rms_null_reason;
  94     json diagnostics = json::object();
  95     json intermediates = json::object();
  96     json images = json::object();  // name -> relative path
  97 
  98     void fail(const std::string& category, const std::string& reason) {
  99         status = "failed";
 100         failure_category = category;
 101         failure_reason = reason;
