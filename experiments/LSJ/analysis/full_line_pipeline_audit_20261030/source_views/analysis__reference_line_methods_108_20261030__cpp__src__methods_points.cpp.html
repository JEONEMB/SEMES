  28 // Visible columns of a target: x0..x1 with the configured step, occluded columns skipped.
  29 std::vector<int> visible_columns(const TargetSpec& t, int* occluded) {
  30     std::vector<int> cols;
  31     int occ = 0;
  32     for (int x = t.x0; x <= t.x1; x += t.column_step) {
  33         if (t.excluded(x)) { ++occ; continue; }
  34         cols.push_back(x);
  35     }
  36     if (occluded) *occluded = occ;
  37     return cols;
  38 }

  40 // Float copy of the band plus a fixed pad (for windows/derivatives around band rows).
  41 struct BandPatch {
  42     cv::Mat f;  // CV_64F
  43     int ox = 0, oy = 0;
  44     double at(int x, int y) const { return f.at<double>(y - oy, x - ox); }
  45 };
  46 
  47 BandPatch band_patch(const cv::Mat& gray, const TargetSpec& t, int pad) {
  48     BandPatch p;
  49     int ax = std::max(0, t.x0 - pad), bx = std::min(gray.cols - 1, t.x1 + pad);
  50     int ay = std::max(0, t.y0 - pad), by = std::min(gray.rows - 1, t.y1 + pad);
  51     gray(cv::Rect(ax, ay, bx - ax + 1, by - ay + 1)).convertTo(p.f, CV_64F);
  52     p.ox = ax; p.oy = ay;
  53     return p;
  54 }

  56 // Nelder-Mead minimiser (fixed standard coefficients).
  57 template <class F>
  58 std::array<double, 3> nelder_mead3(F f, std::array<double, 3> x0, const std::array<double, 3>& step,
  59                                    int max_iter, double tol, int* iters, bool* converged) {
  60     std::array<std::array<double, 3>, 4> s;
  61     std::array<double, 4> v;
  62     s[0] = x0;
  63     for (int i = 0; i < 3; ++i) { s[i + 1] = x0; s[i + 1][i] += step[i]; }
  64     for (int i = 0; i < 4; ++i) v[i] = f(s[i]);
  65     *converged = false;
  66     int it = 0;
  67     for (; it < max_iter; ++it) {
  68         std::array<int, 4> idx = {0, 1, 2, 3};
  69         std::sort(idx.begin(), idx.end(), [&](int a, int b) { return v[a] < v[b]; });
  70         auto s2 = s; auto v2 = v;
  71         for (int i = 0; i < 4; ++i) { s[i] = s2[idx[i]]; v[i] = v2[idx[i]]; }
  72         if (std::fabs(v[3] - v[0]) <= tol * (std::fabs(v[0]) + 1e-30)) { *converged = true; break; }
  73         std::array<double, 3> c = {0, 0, 0};
  74         for (int i = 0; i < 3; ++i) for (int k = 0; k < 3; ++k) c[k] += s[i][k] / 3.0;
  75         auto mix = [&](double a) { std::array<double, 3> r; for (int k = 0; k < 3; ++k) r[k] = c[k] + a * (s[3][k] - c[k]); return r; };
  76         auto xr = mix(-1.0); double fr = f(xr);
  77         if (fr < v[0]) {
  78             auto xe = mix(-2.0); double fe = f(xe);
  79             if (fe < fr) { s[3] = xe; v[3] = fe; } else { s[3] = xr; v[3] = fr; }
  80         } else if (fr < v[2]) {
  81             s[3] = xr; v[3] = fr;
  82         } else {
  83             auto xc = fr < v[3] ? mix(-0.5) : mix(0.5);
  84             double fc = f(xc);
  85             if (fc < std::min(fr, v[3])) { s[3] = xc; v[3] = fc; }
  86             else {
  87                 for (int i = 1; i < 4; ++i) { for (int k = 0; k < 3; ++k) s[i][k] = s[0][k] + 0.5 * (s[i][k] - s[0][k]); v[i] = f(s[i]); }
  88             }
  89         }
  90     }
  91     *iters = it;
  92     int best = (int)(std::min_element(v.begin(), v.end()) - v.begin());
  93     return s[best];
  94 }

 100 // ---------------------------------------------------------------------------------------------
 101 // 01: S01 Fabijanska (2014) Sec. III.B. Gaussian A*exp(-(x-mu)^2/(2 s^2)) fitted with Nelder-Mead to
 102 // Sobel gradient samples along the edge normal (vertical here). Coarse edge: per-column argmax of
 103 // the polarity-signed Sobel response inside the band (replaces the paper's ISODATA + skeleton step).
 104 MethodResult method_profile_gaussian(MethodContext& c) {
 105     MethodResult r;
 106     const TargetSpec& t = c.target;
 107     const json& p = c.params;
 108     const int hw = p.at("half_window").get<int>();
 109     const double min_peak = p.at("min_peak").get<double>();
 110     const int max_iter = p.at("nm_max_iter").get<int>();
 111     const double tol = p.at("nm_tol").get<double>();
 112     const double max_shift = p.at("max_center_shift_px").get<double>();
 113     const int ex_every = p.at("profile_examples_every").get<int>();
 114 
 115     BandPatch bp = band_patch(c.gray, t, hw + 4);
 116     cv::Mat gy;
 117     cv::Sobel(bp.f, gy, CV_64F, 0, 1, 3, 1.0, 0.0, cv::BORDER_REFLECT101);
 118     int occluded = 0;
 119     auto cols = visible_columns(t, &occluded);
 120     std::vector<cv::Point2d> pts;
 121     json cand = json::array(), examples = json::array();
 122     int ci = 0, nm_not_conv = 0;
 123     for (int x : cols) {
 124         int best_y = -1; double best = -1e300;
 125         for (int y = t.y0; y <= t.y1; ++y) {
 126             double v = t.polarity * gy.at<double>(y - bp.oy, x - bp.ox);
 127             if (v > best) { best = v; best_y = y; }
 128         }
 129         json row = {x, best_y};
 130         std::string state;
 131         double A = 0, mu = 0, s = 0; int iters = 0; bool conv = false;
 132         if (best < min_peak) state = excluded("weak_peak");
 133         else if (best_y == t.y0 || best_y == t.y1) state = excluded("peak_at_band_edge");
 134         else {
 135             std::vector<double> ys, fs;
 136             for (int k = -hw; k <= hw; ++k) {
 137                 ys.push_back(best_y + k);
 138                 fs.push_back(t.polarity * gy.at<double>(best_y + k - bp.oy, x - bp.ox));
 139             }
 140             auto sse = [&](const std::array<double, 3>& q) {
 141                 double e = 0, ss = std::fabs(q[2]) + 1e-12;
 142                 for (size_t i = 0; i < ys.size(); ++i) {
 143                     double d = ys[i] - q[1];
 144                     double m = q[0] * std::exp(-d * d / (2 * ss * ss)) - fs[i];
 145                     e += m * m;
 146                 }
 147                 return e;
 148             };
 149             std::array<double, 3> x0 = {best, (double)best_y, (double)(2 * hw + 1)};
 150             std::array<double, 3> st = {0.2 * best, 1.0, 0.5 * (2 * hw + 1)};
 151             auto q = nelder_mead3(sse, x0, st, max_iter, tol, &iters, &conv);
 152             A = q[0]; mu = q[1]; s = std::fabs(q[2]);
 153             if (!conv) ++nm_not_conv;
 154             if (!conv) state = excluded("nelder_mead_not_converged");
 155             else if (A <= 0) state = excluded("non_positive_amplitude");
 156             else if (std::fabs(mu - best_y) > max_shift) state = excluded("center_shift_exceeds_limit");
 157             else { state = "selected"; pts.emplace_back((double)x, mu); }
 158             if (ex_every > 0 && ci % ex_every == 0) {
 159                 examples.push_back({{"x", x}, {"y_samples", ys}, {"sobel_samples", fs}, {"A", A}, {"mu", mu},
 160                                     {"sigma", s}, {"nm_iterations", iters}, {"converged", conv}, {"state", state}});
 161             }
 162         }
 163         row.push_back(state); row.push_back(rnd(mu, 5)); row.push_back(rnd(A, 3)); row.push_back(rnd(s, 4));
 164         cand.push_back(row);
 165         ++ci;
 166     }
 167     r.n_candidates = (int)cols.size();
 168     r.intermediates["columns_fields"] = {"x", "coarse_peak_y", "state", "mu", "A", "sigma"};
 169     r.intermediates["columns"] = cand;
 170     r.intermediates["profile_examples"] = examples;
 171     json fitpts = json::array();
 172     for (auto& q : pts) fitpts.push_back(round_pt(q.x, q.y, 5));
 173     r.intermediates["fit_points"] = fitpts;
 174     r.diagnostics["visible_columns"] = (int)cols.size();
 175     r.diagnostics["occluded_columns_skipped"] = occluded;
 176     r.diagnostics["nelder_mead_not_converged"] = nm_not_conv;
 177     finish_with_points(r, pts, c.fit, "subpixel_edge_points: Gaussian mu on Sobel derivative profile (1 per column)");
 178     return r;
 179 }

 182 // 02: S02 author implementation devernay() (Canny with Devernay sub-pixel correction, chaining,
 183 // hysteresis). Candidate filtering by polarity/orientation and per-column-bin strongest point.
 184 MethodResult method_canny_devernay(MethodContext& c) {
 185     MethodResult r;
 186     const TargetSpec& t = c.target;
 187     const json& p = c.params;
 188     const double sigma = p.at("sigma").get<double>();
 189     const double th_h = p.at("th_h").get<double>(), th_l = p.at("th_l").get<double>();
 190 
 191     cv::Rect reg = t.detector_region(c.gray.size());
 192     cv::Mat f;
 193     c.gray(reg).convertTo(f, CV_64F);
 194     if (!f.isContinuous()) f = f.clone();
 195     double *xs = nullptr, *ys = nullptr; int N = 0, M = 0; int* limits = nullptr;
 196     devernay(&xs, &ys, &N, &limits, &M, f.ptr<double>(), f.cols, f.rows, sigma, th_h, th_l);
 197 
 198     // Gradient of the same Gaussian-smoothed patch, used only for polarity/orientation/strength.
 199     int off = (int)std::ceil(sigma * std::sqrt(2.0 * 3.0 * std::log(10.0)));
 200     cv::Mat g; cv::GaussianBlur(f, g, cv::Size(2 * off + 1, 2 * off + 1), sigma, sigma, cv::BORDER_REPLICATE);
 201     cv::Mat gx = cv::Mat::zeros(g.size(), CV_64F), gyy = cv::Mat::zeros(g.size(), CV_64F);
 202     for (int y = 1; y < g.rows - 1; ++y)
 203         for (int x = 1; x < g.cols - 1; ++x) {
 204             gx.at<double>(y, x) = g.at<double>(y, x + 1) - g.at<double>(y, x - 1);
 205             gyy.at<double>(y, x) = g.at<double>(y + 1, x) - g.at<double>(y - 1, x);
 206         }
 207 
 208     struct Cand { double x, y, strength; int curve; std::string state; };
 209     std::vector<Cand> cands;
 210     int in_region_total = N;
 211     std::map<int, int> best_in_bin;
 212     for (int k = 0; k < M; ++k) {
 213         for (int i = limits[k]; i < limits[k + 1]; ++i) {
 214             double lx = xs[i], ly = ys[i];
 215             double X = lx + reg.x, Y = ly + reg.y;
 216             if (!t.in_band(Y)) continue;  // only band points are candidates
 217             Cand cd{X, Y, 0, k, ""};
 218             double vx = bilinear(gx, lx, ly), vy = bilinear(gyy, lx, ly);
 219             cd.strength = std::hypot(vx, vy);
 220             if (!t.in_x(X)) cd.state = excluded("outside_x_range");
 221             else if (t.excluded(X)) cd.state = excluded("occluded_mask");
 222             else if (t.polarity * vy <= 0) cd.state = excluded("wrong_polarity");
 223             else if (std::fabs(vy) <= std::fabs(vx)) cd.state = excluded("not_near_horizontal_gradient");
 224             else cd.state = "candidate";
 225             cands.push_back(cd);
 226             if (cd.state == "candidate") {
 227                 int bin = (int)std::floor((std::round(X) - t.x0) / (double)t.column_step);
 228                 auto it = best_in_bin.find(bin);
 229                 int idx = (int)cands.size() - 1;
 230                 if (it == best_in_bin.end()) best_in_bin[bin] = idx;
 231                 else if (cands[it->second].strength < cd.strength) {
 232                     cands[it->second].state = excluded("weaker_in_column_bin");
 233                     it->second = idx;
 234                 } else cands[idx].state = excluded("weaker_in_column_bin");
 235             }
 236         }
 237     }
 238     std::free(xs); std::free(ys); std::free(limits);
 239     std::vector<cv::Point2d> pts;
 240     json arr = json::array();
 241     for (auto& cd : cands) {
 242         if (cd.state == "candidate") { cd.state = "selected"; pts.emplace_back(cd.x, cd.y); }
 243         arr.push_back({rnd(cd.x, 4), rnd(cd.y, 4), (int)std::lround(cd.x), (int)std::lround(cd.y), cd.curve, rnd(cd.strength, 3), cd.state});
 244     }
 245     r.n_candidates = (int)cands.size();
 246     r.intermediates["band_points_fields"] = {"x_subpixel", "y_subpixel", "x_rounded", "y_rounded", "curve", "strength", "state"};
 247     r.intermediates["band_points"] = arr;
 248     json fitpts = json::array();
 249     for (auto& q : pts) fitpts.push_back(round_pt(q.x, q.y, 5));
 250     r.intermediates["fit_points"] = fitpts;
 251     r.diagnostics["devernay_points_in_region"] = in_region_total;
 252     r.diagnostics["devernay_curves_in_region"] = M;
 253     r.diagnostics["detector_region_xywh"] = {reg.x, reg.y, reg.width, reg.height};
 254     r.diagnostics["integer_pixel_note"] = "x_rounded/y_rounded = nearest pixel of each Devernay point (the NMS pixel; the correction offset is within +-0.5 px along one axis).";
 255     finish_with_points(r, pts, c.fit, "subpixel_edge_points: Canny-Devernay points, strongest per column bin");
 256     return r;
 257 }

 259 // ---------------------------------------------------------------------------------------------
 260 // 05: S05 Renshaw & Christian (2020). Moments A11 (complex) and A20 from pixel-integrated masks over the
 261 // inscribed unit circle; psi = atan2(Im A11, Re A11) (Eq. 31); Q_E = A20 / (Re A11 cos psi + Im A11 sin psi)
 262 // (Eq. 42); ramp-edge distance l = (1 - w^2 - sqrt((w^2-1)^2 - 2 w^2 Q_E)) / w^2 (Eq. 41), w = k_edge*sigma
 263 // (Eq. 33) in normalised units; location = guess + Np*l/2 (cos psi, sin psi) (Eq. 25).
 264 namespace {
 265 struct ZernikeMasks { int Np; cv::Mat re11, im11, m20; };
 266 const ZernikeMasks& zernike_masks(int Np, int ss) {
 267     static std::mutex mu;
 268     static std::map<std::pair<int, int>, ZernikeMasks> cache;
 269     std::lock_guard<std::mutex> lock(mu);
 270     auto key = std::make_pair(Np, ss);
 271     auto it = cache.find(key);
 272     if (it != cache.end()) return it->second;
 273     ZernikeMasks z; z.Np = Np;
 274     z.re11 = cv::Mat::zeros(Np, Np, CV_64F); z.im11 = z.re11.clone(); z.m20 = z.re11.clone();
 275     int h = Np / 2;
 276     double scale = 2.0 / Np;                       // u_bar = (2/Np)(u - u~), Eq. 1
 277     double da = (scale / ss) * (scale / ss);       // sub-sample area in normalised units
 278     for (int j = -h; j <= h; ++j)
 279         for (int i = -h; i <= h; ++i) {
 280             double sre = 0, sim = 0, s20 = 0;
 281             for (int b = 0; b < ss; ++b)
 282                 for (int a = 0; a < ss; ++a) {
 283                     double u = scale * (i - 0.5 + (a + 0.5) / ss);
 284                     double v = scale * (j - 0.5 + (b + 0.5) / ss);
 285                     double r2 = u * u + v * v;
 286                     if (r2 > 1.0) continue;
 287                     sre += u * da; sim += v * da; s20 += (2 * r2 - 1) * da;
 288                 }
 289             z.re11.at<double>(j + h, i + h) = sre;
 290             z.im11.at<double>(j + h, i + h) = sim;  // = transpose of Re mask (Eq. 23)
 291             z.m20.at<double>(j + h, i + h) = s20;
 292         }
 293     return cache.emplace(key, z).first->second;
 294 }
 295 }  // namespace

 297 MethodResult method_zernike(MethodContext& c) {
 298     MethodResult r;
 299     const TargetSpec& t = c.target;
 300     const json& p = c.params;
 301     const double cb = p.at("coarse_blur_sigma").get<double>();
 302     const double min_g = p.at("min_coarse_grad").get<double>();
 303     const int Np = p.at("Np").get<int>();
 304     const double k_edge = p.at("k_edge").get<double>();
 305     const double blur_px = p.at("blur_sigma_px").get<double>();
 306     const int ss = p.at("supersample").get<int>();
 307     const double max_shift = p.at("max_shift_px").get<double>();
 308     const double max_dev = p.at("max_orientation_dev_deg").get<double>();
 309     const int ex_every = p.at("window_examples_every").get<int>();
 310     const ZernikeMasks& Z = zernike_masks(Np, ss);
 311     const int h = Np / 2;
 312     const double wbar = 2.0 * k_edge * blur_px / Np;
 313 
 314     BandPatch bp = band_patch(c.gray, t, 12);
 315     cv::Mat sm; cv::GaussianBlur(bp.f, sm, cv::Size(0, 0), cb, cb, cv::BORDER_REFLECT101);
 316     int occluded = 0;
 317     auto cols = visible_columns(t, &occluded);
 318     std::vector<cv::Point2d> pts;
 319     json cand = json::array(), examples = json::array();
 320     int ci = 0;
 321     for (int x : cols) {
 322         int best_y = -1; double best = -1e300;
 323         for (int y = t.y0; y <= t.y1; ++y) {
 324             int yy = y - bp.oy, xx = x - bp.ox;
 325             double g = 0.5 * (sm.at<double>(yy + 1, xx) - sm.at<double>(yy - 1, xx));
 326             if (t.polarity * g > best) { best = t.polarity * g; best_y = y; }
 327         }
 328         std::string state;
 329         double psi = 0, Q = 0, l = 0, shift = 0, u = 0, v = 0, are = 0, aim = 0, a20 = 0;
 330         if (best < min_g) state = excluded("weak_coarse_gradient");
 331         else if (best_y == t.y0 || best_y == t.y1) state = excluded("peak_at_band_edge");
 332         else {
 333             cv::Mat win = bp.f(cv::Rect(x - bp.ox - h, best_y - bp.oy - h, Np, Np));
 334             double mean = cv::mean(win)[0];
 335             for (int j = 0; j < Np; ++j)
 336                 for (int i = 0; i < Np; ++i) {
 337                     double I = win.at<double>(j, i) - mean;
 338                     are += Z.re11.at<double>(j, i) * I;
 339                     aim += Z.im11.at<double>(j, i) * I;
 340                     a20 += Z.m20.at<double>(j, i) * I;
 341                 }
 342             psi = std::atan2(aim, are);
 343             double a11p = are * std::cos(psi) + aim * std::sin(psi);
 344             Q = a20 / a11p;
 345             double dev = std::acos(std::max(-1.0, std::min(1.0, t.polarity * std::sin(psi)))) * 180.0 / kPi;
 346             double disc = (wbar * wbar - 1) * (wbar * wbar - 1) - 2 * wbar * wbar * Q;
 347             if (dev > max_dev) state = excluded("orientation_not_matching_target");
 348             else if (disc < 0) state = excluded("ramp_model_no_real_solution");
 349             else {
 350                 l = wbar > 1e-9 ? (1 - wbar * wbar - std::sqrt(disc)) / (wbar * wbar) : Q;
 351                 shift = Np * l / 2.0;
 352                 u = x + shift * std::cos(psi);
 353                 v = best_y + shift * std::sin(psi);
 354                 if (std::fabs(shift) > max_shift) state = excluded("shift_exceeds_limit");
 355                 else { state = "selected"; pts.emplace_back(u, v); }
 356             }
 357             if (ex_every > 0 && ci % ex_every == 0) {
 358                 json w = json::array();
 359                 for (int j = 0; j < Np; ++j) { json row = json::array(); for (int i = 0; i < Np; ++i) row.push_back(win.at<double>(j, i)); w.push_back(row); }
 360                 examples.push_back({{"x", x}, {"coarse_y", best_y}, {"window", w}, {"A11_re", are}, {"A11_im", aim},
 361                                     {"A20", a20}, {"psi_deg", psi * 180 / kPi}, {"Q_E", Q}, {"l_ramp", l},
 362                                     {"l_step_model", Q}, {"shift_px", shift}, {"u", u}, {"v", v}, {"state", state}});
 363             }
 364         }
 365         cand.push_back({x, best_y, state, rnd(psi * 180 / kPi, 4), rnd(Q, 6), rnd(l, 6), rnd(shift, 5), rnd(u, 5), rnd(v, 5),
 366                         rnd(best_y + Np * Q / 2.0 * std::sin(psi), 5)});
 367         ++ci;
 368     }
 369     r.n_candidates = (int)cols.size();
 370     r.intermediates["columns_fields"] = {"x", "coarse_y", "state", "psi_deg", "Q_E", "l_ramp", "shift_px", "u", "v", "v_if_step_model"};
 371     r.intermediates["columns"] = cand;
 372     r.intermediates["window_examples"] = examples;
 373     json masks;
 374     auto m2j = [](const cv::Mat& m) { json a = json::array(); for (int j = 0; j < m.rows; ++j) { json row = json::array(); for (int i = 0; i < m.cols; ++i) row.push_back(m.at<double>(j, i)); a.push_back(row); } return a; };
 375     masks["M11_re"] = m2j(Z.re11); masks["M11_im"] = m2j(Z.im11); masks["M20"] = m2j(Z.m20);
 376     r.intermediates["zernike_masks"] = masks;
 377     json fitpts = json::array();
 378     for (auto& q : pts) fitpts.push_back(round_pt(q.x, q.y, 5));
 379     r.intermediates["fit_points"] = fitpts;
 380     r.diagnostics["w_bar_normalised"] = wbar;
 381     r.diagnostics["visible_columns"] = (int)cols.size();
 382     r.diagnostics["occluded_columns_skipped"] = occluded;
 383     finish_with_points(r, pts, c.fit, "subpixel_edge_points: Zernike ramp-edge location (2D, 1 per column)");
 384     return r;
 385 }

 387 // ---------------------------------------------------------------------------------------------
 388 // 06: S06 Trujillo-Pino et al. (2013), partial area effect, second-order (y = a + b x + c x^2) basic
 389 // model on the raw image. Window 3 x (2m+1). Column sums S_j = B(2m+1) + (A-B)(a + b j + c(j^2+1/12) + m + 1/2)
 390 // give c = (S_L+S_R-2S_M)/(2(A-B)), b = (S_R-S_L)/(2(A-B)), a = (2S_M-(A+B)(2m+1))/(2(A-B)) - c/12.
 391 // A/B: mean of the 3 pixels of the window's first/last row (A = side of smaller y).
 392 MethodResult method_pae(MethodContext& c) {
 393     MethodResult r;
 394     const TargetSpec& t = c.target;
 395     const json& p = c.params;
 396     const int m = p.at("half_height").get<int>();
 397     const double min_g = p.at("min_grad").get<double>();
 398     const double min_contrast = p.at("min_contrast").get<double>();
 399     const double max_a = p.at("max_abs_a_px").get<double>();
 400 
 401     BandPatch bp = band_patch(c.gray, t, m + 4);
 402     int occluded = 0;
 403     auto cols = visible_columns(t, &occluded);
 404     std::vector<cv::Point2d> pts;
 405     json cand = json::array(), examples = json::array();
 406     auto I = [&](int x, int y) { return bp.at(x, y); };
 407     auto fy = [&](int x, int y) { return 0.5 * (I(x, y + 1) - I(x, y - 1)); };
 408     auto fx = [&](int x, int y) { return 0.5 * (I(x + 1, y) - I(x - 1, y)); };
 409     int ci = 0;
 410     for (int x : cols) {
 411         int best_y = -1; double best = -1;
 412         for (int y = t.y0; y <= t.y1; ++y) {
 413             double gy = fy(x, y), gx = fx(x, y);
 414             // Edge pixel rule for horizontal edges (paper / Method.md): |fy|>|fx| and local max of |fy| in y.
 415             if (t.polarity * gy <= 0 || std::fabs(gy) <= std::fabs(gx)) continue;
 416             if (std::fabs(fy(x, y - 1)) > std::fabs(gy) || std::fabs(fy(x, y + 1)) > std::fabs(gy)) continue;
 417             if (std::fabs(gy) > best) { best = std::fabs(gy); best_y = y; }
 418         }
 419         std::string state;
 420         double a = 0, b = 0, cc = 0, A = 0, B = 0;
 421         if (best_y < 0) state = excluded("no_edge_pixel");
 422         else if (best < min_g) state = excluded("weak_gradient");
 423         else {
 424             double S[3] = {0, 0, 0};
 425             for (int j = -1; j <= 1; ++j) for (int k = -m; k <= m; ++k) S[j + 1] += I(x + j, best_y + k);
 426             A = (I(x - 1, best_y - m) + I(x, best_y - m) + I(x + 1, best_y - m)) / 3.0;
 427             B = (I(x - 1, best_y + m) + I(x, best_y + m) + I(x + 1, best_y + m)) / 3.0;
 428             if (t.polarity * (B - A) < min_contrast) state = excluded("low_contrast_A_B");
 429             else {
 430                 double den = 2 * (A - B);
 431                 cc = (S[0] + S[2] - 2 * S[1]) / den;
 432                 b = (S[2] - S[0]) / den;
 433                 a = (2 * S[1] - (A + B) * (2 * m + 1)) / den - cc / 12.0;
 434                 if (std::fabs(a) > max_a) state = excluded("a_exceeds_limit");
 435                 else { state = "selected"; pts.emplace_back((double)x, best_y + a); }
 436             }
 437             if (ci % 100 == 0) {
 438                 json w = json::array();
 439                 for (int k = -m; k <= m; ++k) w.push_back({I(x - 1, best_y + k), I(x, best_y + k), I(x + 1, best_y + k)});
 440                 examples.push_back({{"x", x}, {"edge_pixel_y", best_y}, {"window_rows_top_to_bottom", w}, {"S_L", S[0]}, {"S_M", S[1]},
 441                                     {"S_R", S[2]}, {"A", A}, {"B", B}, {"a", a}, {"b", b}, {"c", cc}, {"state", state}});
 442             }
 443         }
 444         cand.push_back({x, best_y, state, rnd(a, 5), rnd(b, 5), rnd(cc, 5), rnd(A, 2), rnd(B, 2)});
 445         ++ci;
 446     }
 447     r.n_candidates = (int)cols.size();
 448     r.intermediates["columns_fields"] = {"x", "edge_pixel_y", "state", "a", "b", "c", "A", "B"};
 449     r.intermediates["columns"] = cand;
 450     r.intermediates["window_examples"] = examples;
 451     json fitpts = json::array();
 452     for (auto& q : pts) fitpts.push_back(round_pt(q.x, q.y, 5));
 453     r.intermediates["fit_points"] = fitpts;
 454     r.diagnostics["window"] = "3x" + std::to_string(2 * m + 1);
 455     r.diagnostics["visible_columns"] = (int)cols.size();
 456     r.diagnostics["occluded_columns_skipped"] = occluded;
 457     finish_with_points(r, pts, c.fit, "subpixel_edge_points: PAE second-order a at column centre (1 per column)");
 458     return r;
 459 }
