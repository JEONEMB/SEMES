   1 // 03: Canny -> standard Hough transform (S03) -> select the target boundary -> refit on Canny pixels.
   2 // Hough theta is the angle of the line NORMAL; the image line angle is theta - 90 deg.
   3 #include "common.hpp"
   4 #include "methods.hpp"
   5 
   6 #include <opencv2/imgproc.hpp>
   7 
   8 #include <cmath>
   9 
  10 namespace lm {
  11 
  12 MethodResult method_canny_hough(MethodContext& c) {
  13     MethodResult r;
  14     const TargetSpec& t = c.target;
  15     const json& p = c.params;
  16     const double kDeg = 3.14159265358979323846 / 180.0;
  17     cv::Rect reg = t.detector_region(c.gray.size());
  18     cv::Mat src = c.gray(reg), blur;
  19     cv::GaussianBlur(src, blur, cv::Size(0, 0), p.at("blur_sigma").get<double>(), 0, cv::BORDER_REFLECT101);
  20     cv::Mat edges;
  21     cv::Canny(blur, edges, p.at("canny_low").get<double>(), p.at("canny_high").get<double>(),
  22               p.at("canny_aperture").get<int>(), p.at("l2gradient").get<bool>());
  23     cv::Mat gx, gy;
  24     cv::Sobel(blur, gx, CV_32F, 1, 0, 3);
  25     cv::Sobel(blur, gy, CV_32F, 0, 1, 3);
  26     // Vote only with Canny pixels of the target polarity, inside band / x range / visible mask.
  27     cv::Mat filt = cv::Mat::zeros(edges.size(), CV_8U);
  28     int n_canny = 0, n_filt = 0;
  29     std::vector<cv::Point2d> edge_px;
  30     for (int y = 0; y < edges.rows; ++y)
  31         for (int x = 0; x < edges.cols; ++x) {
  32             if (!edges.at<uchar>(y, x)) continue;
  33             ++n_canny;
  34             double X = x + reg.x, Y = y + reg.y;
  35             float vx = gx.at<float>(y, x), vy = gy.at<float>(y, x);
  36             if (!t.in_band(Y) || !t.in_x(X) || t.excluded(X)) continue;
  37             if (t.polarity * vy <= 0 || std::fabs(vy) <= std::fabs(vx)) continue;
  38             filt.at<uchar>(y, x) = 255;
  39             edge_px.emplace_back(X, Y);
  40             ++n_filt;
  41         }
  42     c.sink.save_png("canny_edges", edges, reg.tl(), r.images);
  43     c.sink.save_png("canny_polarity_band_filtered", filt, reg.tl(), r.images);
  44 
  45     std::vector<cv::Vec3f> lines;
  46     cv::HoughLines(filt, lines, p.at("hough_rho").get<double>(), p.at("hough_theta_deg").get<double>() * kDeg,
  47                    p.at("hough_threshold").get<int>(), 0, 0, p.at("theta_min_deg").get<double>() * kDeg,
  48                    p.at("theta_max_deg").get<double>() * kDeg);
  49     r.n_candidates = (int)lines.size();
  50     int max_saved = p.at("max_candidates_saved").get<int>();
  51     json cand = json::array();
  52     int sel = -1;
  53     Line hline;
  54     for (size_t i = 0; i < lines.size(); ++i) {
  55         double rho = lines[i][0], th = lines[i][1], votes = lines[i][2];
  56         double ct = std::cos(th), st = std::sin(th);
  57         double rho_o = rho + reg.x * ct + reg.y * st;  // back to original coordinates
  58         Line l = normalize_line(rho_o * ct, rho_o * st, -st, ct);
  59         double ya = (rho_o - t.x0 * ct) / st, yb = (rho_o - t.x1 * ct) / st;
  60         std::string state;
  61         if (!t.in_band(ya) || !t.in_band(yb)) state = "excluded:outside_band";
  62         else if (sel < 0) { state = "selected"; sel = (int)i; hline = l; }
  63         else state = "not_selected:fewer_votes";
  64         if ((int)i < max_saved || state == "selected")
  65             cand.push_back({{"rank", (int)i}, {"rho_original_px", rho_o}, {"theta_normal_deg", th / kDeg}, {"votes", votes},
  66                             {"image_angle_deg", line_angle_deg(l)}, {"y_at_x0", ya}, {"y_at_x1", yb}, {"state", state}});
  67     }
  68     r.intermediates["hough_candidates"] = cand;
  69     r.diagnostics["canny_pixels_in_region"] = n_canny;
  70     r.diagnostics["voting_pixels_after_polarity_band_mask"] = n_filt;
  71     r.diagnostics["hough_lines_returned"] = (int)lines.size();
  72     r.diagnostics["accumulator_note"] = "OpenCV HoughLines does not expose the accumulator; votes per returned line are saved.";
  73     if (sel < 0) {
  74         r.fail("detection_failed", lines.empty() ? "no Hough line above vote threshold" : "no Hough line inside the target band");
  75         return r;
  76     }
  77     r.diagnostics["hough_selected_angle_deg"] = line_angle_deg(hline);
  78     r.diagnostics["hough_selected_line"] = line_json(hline);
  79     // Refit: Canny pixels (integer) within refit distance of the selected Hough line.
  80     double dmax = p.at("refit_dist_px").get<double>();
  81     std::vector<cv::Point2d> near;
  82     json pts = json::array();
  83     for (auto& q : edge_px) {
  84         bool keep = std::fabs(signed_distance(hline, q.x, q.y)) <= dmax;
  85         if (keep) near.push_back(q);
  86         pts.push_back({(int)q.x, (int)q.y, keep ? 1 : 0});
  87     }
  88     r.intermediates["voting_pixels_fields"] = {"x", "y", "within_refit_dist"};
  89     r.intermediates["voting_pixels"] = pts;
  90     json fitpts = json::array();
  91     for (auto& q : near) fitpts.push_back(round_pt(q.x, q.y, 1));
  92     r.intermediates["fit_points"] = fitpts;
  93     finish_with_points(r, near, c.fit, "canny_integer_pixels within refit_dist of the selected Hough line");
  94     return r;
  95 }
