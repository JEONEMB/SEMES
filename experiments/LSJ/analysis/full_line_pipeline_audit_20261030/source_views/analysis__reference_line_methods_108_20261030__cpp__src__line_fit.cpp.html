  13 cv::Rect TargetSpec::detector_region(const cv::Size& image) const {
  14     int ax = x0 - margin, bx = x1 + margin, ay = y0 - margin, by = y1 + margin;
  15     if (kind == "qr") {
  16         ax = roi.x - margin; bx = roi.x + roi.width - 1 + margin;
  17         ay = roi.y - margin; by = roi.y + roi.height - 1 + margin;
  18     }
  19     ax = std::max(ax, 0); ay = std::max(ay, 0);
  20     bx = std::min(bx, image.width - 1); by = std::min(by, image.height - 1);
  21     return cv::Rect(ax, ay, bx - ax + 1, by - ay + 1);
  22 }

  24 Line normalize_line(double px, double py, double dx, double dy) {
  25     double n = std::hypot(dx, dy);
  26     Line l;
  27     l.px = px; l.py = py;
  28     if (n <= 0) { l.dx = 1; l.dy = 0; return l; }
  29     dx /= n; dy /= n;
  30     // x-increasing direction; a vertical line points to -y so the angle is -90 (range [-90, 90)).
  31     if (dx < 0 || (dx == 0 && dy > 0)) { dx = -dx; dy = -dy; }
  32     if (dx == 0) dy = -1;
  33     l.dx = dx; l.dy = dy;
  34     return l;
  35 }

  37 double line_angle_deg(const Line& l) {
  38     double a = std::atan2(l.dy, l.dx) * 180.0 / kPi;
  39     if (a >= 90.0) a -= 180.0;
  40     if (a < -90.0) a += 180.0;
  41     return a;
  42 }

  44 double signed_distance(const Line& l, double x, double y) {
  45     return -(x - l.px) * l.dy + (y - l.py) * l.dx;
  46 }

  48 json line_json(const Line& l) {
  49     json j;
  50     j["point_xy"] = {l.px, l.py};
  51     j["direction_xy"] = {l.dx, l.dy};
  52     if (std::fabs(l.dx) > 1e-12) {
  53         double m = l.dy / l.dx;
  54         j["slope_dy_dx"] = m;
  55         j["intercept_y_px"] = l.py - m * l.px;
  56     } else {
  57         j["slope_dy_dx"] = nullptr;
  58         j["intercept_y_px"] = nullptr;
  59         j["slope_null_reason"] = "vertical line: dx=0";
  60     }
  61     // a*x + b*y + c = 0 with (a, b) the unit normal (-dy, dx).
  62     double a = -l.dy, b = l.dx, c = -(a * l.px + b * l.py);
  63     j["general_form_abc"] = {a, b, c};
  64     j["image_angle_deg"] = line_angle_deg(l);
  65     return j;
  66 }

  68 json line_draw_endpoints(const Line& l, double xa, double xb) {
  69     if (std::fabs(l.dx) < 1e-12) return nullptr;
  70     double m = l.dy / l.dx, b = l.py - m * l.px;
  71     return json::array({json::array({xa, m * xa + b}), json::array({xb, m * xb + b})});
  72 }

  74 Line huber_line(const std::vector<cv::Point2d>& pts, const std::vector<char>* use, const FitParams& p) {
  75     std::vector<double> w(pts.size(), 1.0);
  76     if (use)
  77         for (size_t i = 0; i < pts.size(); ++i) w[i] = (*use)[i] ? 1.0 : 0.0;
  78     std::vector<char> active(pts.size());
  79     for (size_t i = 0; i < pts.size(); ++i) active[i] = w[i] > 0;
  80     // Shift for conditioning; the result is shifted back.
  81     double ox = 0, oy = 0; int n = 0;
  82     for (size_t i = 0; i < pts.size(); ++i) if (active[i]) { ox += pts[i].x; oy += pts[i].y; ++n; }
  83     if (n == 0) return Line();
  84     ox /= n; oy /= n;
  85     Line line;
  86     double prev_angle = std::numeric_limits<double>::quiet_NaN(), prev_cx = 0, prev_cy = 0;
  87     for (int it = 0; it < p.irls_max_iter; ++it) {
  88         double sw = 0, sx = 0, sy = 0;
  89         for (size_t i = 0; i < pts.size(); ++i) {
  90             if (!active[i]) continue;
  91             sw += w[i]; sx += w[i] * (pts[i].x - ox); sy += w[i] * (pts[i].y - oy);
  92         }
  93         double cx = sx / sw, cy = sy / sw;
  94         double sxx = 0, syy = 0, sxy = 0;
  95         for (size_t i = 0; i < pts.size(); ++i) {
  96             if (!active[i]) continue;
  97             double u = pts[i].x - ox - cx, v = pts[i].y - oy - cy;
  98             sxx += w[i] * u * u; syy += w[i] * v * v; sxy += w[i] * u * v;
  99         }
 100         double angle = 0.5 * std::atan2(2 * sxy, sxx - syy);
 101         line = normalize_line(cx + ox, cy + oy, std::cos(angle), std::sin(angle));
 102         bool converged = it > 0 && std::fabs(angle - prev_angle) < p.irls_tol &&
 103                          std::fabs(cx - prev_cx) < 1e-9 && std::fabs(cy - prev_cy) < 1e-9;
 104         prev_angle = angle; prev_cx = cx; prev_cy = cy;
 105         if (converged) break;
 106         for (size_t i = 0; i < pts.size(); ++i) {
 107             if (!active[i]) continue;
 108             double r = std::fabs(signed_distance(line, pts[i].x, pts[i].y));
 109             w[i] = r <= p.huber_c ? 1.0 : p.huber_c / r;
 110         }
 111     }
 112     return line;
 113 }

 115 FitResult robust_fit(const std::vector<cv::Point2d>& pts, const FitParams& p) {
 116     FitResult f;
 117     f.n_all = static_cast<int>(pts.size());
 118     if (f.n_all < p.min_points) {
 119         f.reason = "too few points for fit: " + std::to_string(f.n_all) + " < " + std::to_string(p.min_points);
 120         return f;
 121     }
 122     Line line = huber_line(pts, nullptr, p);
 123     f.initial_angle_deg = line_angle_deg(line);
 124     std::vector<char> in(pts.size(), 1);
 125     for (int it = 0; it < p.trim_iterations; ++it) {
 126         int count = 0;
 127         for (size_t i = 0; i < pts.size(); ++i) {
 128             in[i] = std::fabs(signed_distance(line, pts[i].x, pts[i].y)) <= p.inlier_px;
 129             count += in[i];
 130         }
 131         if (count < p.min_points) {
 132             f.reason = "too few inliers after trimming: " + std::to_string(count);
 133             return f;
 134         }
 135         line = huber_line(pts, &in, p);
 136     }
 137     double s_in = 0, s_all = 0, mx = 0; int n_in = 0;
 138     for (size_t i = 0; i < pts.size(); ++i) {
 139         double r = signed_distance(line, pts[i].x, pts[i].y);
 140         s_all += r * r;
 141         in[i] = std::fabs(r) <= p.inlier_px;
 142         if (in[i]) { s_in += r * r; ++n_in; mx = std::max(mx, std::fabs(r)); }
 143     }
 144     f.ok = true;
 145     f.line = line;
 146     f.inlier = in;
 147     f.n_inlier = n_in;
 148     f.rms_inlier = n_in ? std::sqrt(s_in / n_in) : 0;
 149     f.rms_all = std::sqrt(s_all / pts.size());
 150     f.max_abs_inlier = mx;
 151     return f;
 152 }

 154 FitParams fit_params_for(const json& common, const std::string& kind) {
 155     const json& f = common.at("fit");
 156     FitParams p;
 157     p.huber_c = f.at("huber_c_px").get<double>();
 158     p.irls_max_iter = f.at("irls_max_iter").get<int>();
 159     p.irls_tol = f.at("irls_tol").get<double>();
 160     p.trim_iterations = f.at("trim_iterations").get<int>();
 161     p.inlier_px = f.at("inlier_px").at(kind).get<double>();
 162     p.min_points = f.at("min_points").at(kind).get<int>();
 163     return p;
 164 }

 166 void finish_with_points(MethodResult& r, const std::vector<cv::Point2d>& pts, const FitParams& fp,
 167                         const std::string& support_type) {
 168     r.support_type = support_type;
 169     FitResult f = robust_fit(pts, fp);
 170     r.diagnostics["fit_input_points"] = f.n_all;
 171     if (!f.ok) {
 172         r.fail("fit_failed", f.reason);
 173         return;
 174     }
 175     r.status = "ok";
 176     r.has_line = true;
 177     r.line = f.line;
 178     r.n_selected = f.n_inlier;
 179     r.has_rms = true;
 180     r.fit_rms = f.rms_inlier;
 181     r.diagnostics["fit_inlier_points"] = f.n_inlier;
 182     r.diagnostics["fit_rms_all_input_px"] = f.rms_all;
 183     r.diagnostics["fit_max_abs_inlier_residual_px"] = f.max_abs_inlier;
 184     r.diagnostics["huber_all_points_angle_before_trim_deg"] = f.initial_angle_deg;
 185     double tmin = 1e300, tmax = -1e300;
 186     for (size_t i = 0; i < pts.size(); ++i) {
 187         if (!f.inlier[i]) continue;
 188         double t = (pts[i].x - f.line.px) * f.line.dx + (pts[i].y - f.line.py) * f.line.dy;
 189         tmin = std::min(tmin, t); tmax = std::max(tmax, t);
 190     }
 191     r.has_endpoints = true;
 192     r.ex1 = f.line.px + tmin * f.line.dx; r.ey1 = f.line.py + tmin * f.line.dy;
 193     r.ex2 = f.line.px + tmax * f.line.dx; r.ey2 = f.line.py + tmax * f.line.dy;
 194     // Inlier flags let the report colour fitted support points.
 195     json flags = json::array();
 196     for (char c : f.inlier) flags.push_back(c ? 1 : 0);
 197     r.intermediates["fit_points_inlier_flag"] = flags;
 198 }

 200 double bilinear(const cv::Mat& f, double x, double y) {
 201     int x0 = (int)std::floor(x), y0 = (int)std::floor(y);
 202     x0 = std::max(0, std::min(x0, f.cols - 2));
 203     y0 = std::max(0, std::min(y0, f.rows - 2));
 204     double ax = x - x0, ay = y - y0;
 205     auto at = [&](int yy, int xx) {
 206         return f.depth() == CV_64F ? f.at<double>(yy, xx) : (double)f.at<float>(yy, xx);
 207     };
 208     return (1 - ay) * ((1 - ax) * at(y0, x0) + ax * at(y0, x0 + 1)) + ay * ((1 - ax) * at(y0 + 1, x0) + ax * at(y0 + 1, x0 + 1));
 209 }

 211 double rnd(double v, int digits) {
 212     double s = std::pow(10.0, digits);
 213     return std::round(v * s) / s;
 214 }
