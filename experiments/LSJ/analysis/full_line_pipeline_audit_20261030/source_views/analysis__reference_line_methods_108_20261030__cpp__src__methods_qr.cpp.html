  18 const char* kTwoPointRms = "two-point line: residual is 0 by construction and is not accuracy evidence";

  20 bool touches_border(const cv::Rect& b, const cv::Size& s) {
  21     return b.x <= 0 || b.y <= 0 || b.x + b.width >= s.width || b.y + b.height >= s.height;
  22 }

  24 void two_point_result(MethodResult& r, const cv::Point2d& a, const cv::Point2d& b, const std::string& support) {
  25     r.status = "ok";
  26     r.has_line = true;
  27     r.line = normalize_line(a.x, a.y, b.x - a.x, b.y - a.y);
  28     r.has_endpoints = true;
  29     r.ex1 = a.x; r.ey1 = a.y; r.ex2 = b.x; r.ey2 = b.y;
  30     r.n_selected = 2;
  31     r.support_type = support;
  32     r.has_rms = false;
  33     r.rms_null_reason = kTwoPointRms;
  34 }

  37 MethodResult method_qr_contour(MethodContext& c) {
  38     MethodResult r;
  39     const TargetSpec& t = c.target;
  40     const json& p = c.params;
  41     cv::Mat crop = c.gray(t.roi).clone(), bin;
  42     double thr = cv::threshold(crop, bin, 0, 255, cv::THRESH_BINARY | cv::THRESH_OTSU);
  43     c.sink.save_png("otsu_binary", bin, t.roi.tl(), r.images);
  44     std::vector<std::vector<cv::Point>> contours;
  45     cv::findContours(bin.clone(), contours, cv::RETR_EXTERNAL, cv::CHAIN_APPROX_NONE);
  46     r.diagnostics["otsu_threshold"] = thr;
  47     r.n_candidates = (int)contours.size();
  48     int best = -1; double best_area = 0;
  49     json cj = json::array();
  50     for (size_t i = 0; i < contours.size(); ++i) {
  51         cv::Rect b = cv::boundingRect(contours[i]);
  52         double area = std::fabs(cv::contourArea(contours[i]));
  53         bool border = touches_border(b, crop.size());
  54         cj.push_back({{"index", (int)i}, {"area", area}, {"bbox_xywh_original", {b.x + t.roi.x, b.y + t.roi.y, b.width, b.height}}, {"touches_roi_border", border}});
  55         if (!border && area > best_area) { best_area = area; best = (int)i; }
  56     }
  57     r.intermediates["external_contours"] = cj;
  58     if (best < 0) { r.fail("detection_failed", "no external bright contour inside the ROI (all touch the ROI border)"); return r; }
  59     const auto& ct = contours[best];
  60     std::vector<cv::Point> poly;
  61     cv::approxPolyDP(ct, poly, p.at("approx_eps_frac").get<double>() * cv::arcLength(ct, true), true);
  62     json pj = json::array();
  63     for (auto& q : poly) pj.push_back({q.x + t.roi.x, q.y + t.roi.y});
  64     r.intermediates["polygon_vertices"] = pj;
  65     r.intermediates["selected_contour_index"] = best;
  66     json full = json::array();
  67     for (auto& q : ct) full.push_back({q.x + t.roi.x, q.y + t.roi.y});
  68     r.intermediates["selected_contour"] = full;
  69     if (poly.size() != 4) {
  70         r.fail("corner_not_identified", "approxPolyDP gave " + std::to_string(poly.size()) + " vertices, expected 4");
  71         return r;
  72     }
  73     auto tl = *std::min_element(poly.begin(), poly.end(), [](const cv::Point& a, const cv::Point& b) { return a.x + a.y < b.x + b.y; });
  74     auto tr = *std::max_element(poly.begin(), poly.end(), [](const cv::Point& a, const cv::Point& b) { return a.x - a.y < b.x - b.y; });
  75     auto nearest = [&](const cv::Point& v) {
  76         int k = 0; double d = 1e300;
  77         for (size_t i = 0; i < ct.size(); ++i) { double e = cv::norm(ct[i] - v); if (e < d) { d = e; k = (int)i; } }
  78         return k;
  79     };
  80     int itl = nearest(tl), itr = nearest(tr), n = (int)ct.size();
  81     // Two contour paths join TL and TR; the top side is the one with the smaller mean y.
  82     std::vector<cv::Point> p1, p2;
  83     for (int k = itl;; k = (k + 1) % n) { p1.push_back(ct[k]); if (k == itr) break; }
  84     for (int k = itl;; k = (k - 1 + n) % n) { p2.push_back(ct[k]); if (k == itr) break; }
  85     auto mean_y = [](const std::vector<cv::Point>& v) { double s = 0; for (auto& q : v) s += q.y; return s / std::max<size_t>(1, v.size()); };
  86     const auto& top = mean_y(p1) < mean_y(p2) ? p1 : p2;
  87     double trim = p.at("corner_trim_frac").get<double>() * (tr.x - tl.x);
  88     std::vector<cv::Point2d> pts;
  89     json tj = json::array();
  90     for (auto& q : top) {
  91         bool keep = q.x >= tl.x + trim && q.x <= tr.x - trim;
  92         tj.push_back({q.x + t.roi.x, q.y + t.roi.y, keep ? 1 : 0});
  93         if (keep) pts.emplace_back(q.x + t.roi.x, q.y + t.roi.y);
  94     }
  95     r.intermediates["top_run_fields"] = {"x", "y", "used_after_corner_trim"};
  96     r.intermediates["top_run"] = tj;
  97     r.diagnostics["top_corners_from_polygon"] = {{tl.x + t.roi.x, tl.y + t.roi.y}, {tr.x + t.roi.x, tr.y + t.roi.y}};
  98     r.diagnostics["contour_offset_note"] = "contour points are centres of the outermost bright pixels; the physical boundary lies about 0.5 px above (constant offset, no slope effect)";
  99     json fitpts = json::array();
 100     for (auto& q : pts) fitpts.push_back(round_pt(q.x, q.y, 1));
 101     r.intermediates["fit_points"] = fitpts;
 102     finish_with_points(r, pts, c.fit, "contour_integer_pixels: top side of the Otsu outer contour, corners trimmed");
 103     return r;
 104 }

 106 MethodResult method_qr_corners(MethodContext& c) {
 107     MethodResult r;
 108     const TargetSpec& t = c.target;
 109     const json& p = c.params;
 110     cv::Mat crop = c.gray(t.roi).clone();
 111     std::vector<cv::Point2f> corners;
 112     cv::goodFeaturesToTrack(crop, corners, p.at("max_corners").get<int>(), p.at("quality_level").get<double>(),
 113                             p.at("min_distance").get<double>(), cv::noArray(), p.at("block_size").get<int>(),
 114                             p.at("use_harris").get<bool>());
 115     r.n_candidates = (int)corners.size();
 116     const int d = p.at("quad_offset_px").get<int>(), q = p.at("quad_half_px").get<int>();
 117     const double minc = p.at("min_quad_contrast").get<double>();
 118     auto box = [&](double cx, double cy) {
 119         int x0 = (int)std::lround(cx) - q, y0 = (int)std::lround(cy) - q;
 120         if (x0 < 0 || y0 < 0 || x0 + 2 * q + 1 > crop.cols || y0 + 2 * q + 1 > crop.rows) return -1.0;
 121         return cv::mean(crop(cv::Rect(x0, y0, 2 * q + 1, 2 * q + 1)))[0];
 122     };
 123     json cj = json::array();
 124     int itl = -1, itr = -1;
 125     std::vector<std::string> kind(corners.size());
 126     for (size_t i = 0; i < corners.size(); ++i) {
 127         double x = corners[i].x, y = corners[i].y;
 128         double ul = box(x - d, y - d), ur = box(x + d, y - d), ll = box(x - d, y + d), lr = box(x + d, y + d);
 129         std::string k = "other";
 130         if (ul >= 0 && ur >= 0 && ll >= 0 && lr >= 0) {
 131             if (lr - std::max({ul, ur, ll}) >= minc) k = "outer_top_left_pattern";
 132             else if (ll - std::max({ul, ur, lr}) >= minc) k = "outer_top_right_pattern";
 133         } else k = "too_close_to_roi_border";
 134         kind[i] = k;
 135         if (k == "outer_top_left_pattern" && (itl < 0 || y < corners[itl].y - 0.5 || (std::fabs(y - corners[itl].y) <= 0.5 && x < corners[itl].x))) itl = (int)i;
 136         if (k == "outer_top_right_pattern" && (itr < 0 || y < corners[itr].y - 0.5 || (std::fabs(y - corners[itr].y) <= 0.5 && x > corners[itr].x))) itr = (int)i;
 137         cj.push_back({{"xy", {x + t.roi.x, y + t.roi.y}}, {"quadrant_means_ul_ur_ll_lr", {ul, ur, ll, lr}}, {"pattern", k}});
 138     }
 139     for (size_t i = 0; i < corners.size(); ++i) {
 140         std::string st = (int)i == itl ? "selected_top_left" : (int)i == itr ? "selected_top_right" : "not_selected";
 141         cj[i]["state"] = st;
 142     }
 143     r.intermediates["corner_candidates"] = cj;
 144     if (itl < 0 || itr < 0) {
 145         r.fail("corner_not_identified", std::string("outer border corner pattern not found: ") + (itl < 0 ? "top-left " : "") + (itr < 0 ? "top-right" : ""));
 146         return r;
 147     }
 148     cv::Point2f a0 = corners[itl], b0 = corners[itr];
 149     if (b0.x - a0.x < 0.5 * crop.cols || std::fabs(b0.y - a0.y) > 15) {
 150         r.fail("corner_not_identified", "identified top corners are not a plausible outer-border top pair");
 151         r.intermediates["rejected_pair"] = {{a0.x + t.roi.x, a0.y + t.roi.y}, {b0.x + t.roi.x, b0.y + t.roi.y}};
 152         return r;
 153     }
 154     std::vector<cv::Point2f> ref = {a0, b0};
 155     int hw = p.at("subpix_half_win").get<int>();
 156     cv::cornerSubPix(crop, ref, cv::Size(hw, hw), cv::Size(-1, -1),
 157                      cv::TermCriteria(cv::TermCriteria::COUNT | cv::TermCriteria::EPS, p.at("subpix_max_iter").get<int>(), p.at("subpix_eps").get<double>()));
 158     double s1 = cv::norm(ref[0] - a0), s2 = cv::norm(ref[1] - b0);
 159     r.intermediates["top_corners"] = {
 160         {{"name", "top_left"}, {"initial_xy", {a0.x + t.roi.x, a0.y + t.roi.y}}, {"refined_xy", {ref[0].x + t.roi.x, ref[0].y + t.roi.y}}, {"shift_px", s1}},
 161         {{"name", "top_right"}, {"initial_xy", {b0.x + t.roi.x, b0.y + t.roi.y}}, {"refined_xy", {ref[1].x + t.roi.x, ref[1].y + t.roi.y}}, {"shift_px", s2}}};
 162     double max_shift = p.at("max_refine_shift_px").get<double>();
 163     if (s1 > max_shift || s2 > max_shift) {
 164         r.fail("corner_not_identified", "cornerSubPix moved a corner farther than max_refine_shift_px");
 165         return r;
 166     }
 167     two_point_result(r, cv::Point2d(ref[0].x + t.roi.x, ref[0].y + t.roi.y), cv::Point2d(ref[1].x + t.roi.x, ref[1].y + t.roi.y),
 168                      "two refined outer-border corners (top-left, top-right)");
 169     return r;
 170 }

 173 std::vector<cv::Point2d> quad_order(const std::vector<cv::Point>& q) {
 174     std::vector<cv::Point2d> p(q.begin(), q.end());
 175     cv::Point2d c(0, 0);
 176     for (auto& v : p) c += v;
 177     c *= 0.25;
 178     std::sort(p.begin(), p.end(), [&](const cv::Point2d& a, const cv::Point2d& b) {
 179         return std::atan2(a.y - c.y, a.x - c.x) < std::atan2(b.y - c.y, b.x - c.x);
 180     });
 181     int k = 0;
 182     for (int i = 1; i < 4; ++i) if (p[i].x + p[i].y < p[k].x + p[k].y) k = i;
 183     std::rotate(p.begin(), p.begin() + k, p.end());
 184     return p;
 185 }
 186 bool diag_intersection(const std::vector<cv::Point2d>& q, cv::Point2d& out) {
 187     cv::Point2d d1 = q[2] - q[0], d2 = q[3] - q[1], rhs = q[1] - q[0];
 188     double det = d1.x * (-d2.y) - (-d2.x) * d1.y;
 189     if (std::fabs(det) < 1e-12) return false;
 190     double tt = (rhs.x * (-d2.y) - (-d2.x) * rhs.y) / det;
 191     double ss = (d1.x * rhs.y - d1.y * rhs.x) / det;
 192     if (!(tt > 0 && tt < 1 && ss > 0 && ss < 1)) return false;
 193     out = q[0] + tt * d1;
 194     return true;
 195 }

 198 MethodResult method_qr_finders(MethodContext& c) {
 199     MethodResult r;
 200     const TargetSpec& t = c.target;
 201     const json& p = c.params;
 202     cv::Mat crop = c.gray(t.roi).clone(), mask;
 203     double thr = cv::threshold(crop, mask, 0, 255, cv::THRESH_BINARY_INV | cv::THRESH_OTSU);
 204     c.sink.save_png("otsu_binary_inv", mask, t.roi.tl(), r.images);
 205     std::vector<std::vector<cv::Point>> contours;
 206     std::vector<cv::Vec4i> hier;
 207     cv::findContours(mask.clone(), contours, hier, cv::RETR_TREE, cv::CHAIN_APPROX_SIMPLE);
 208     r.diagnostics["otsu_threshold"] = thr;
 209     r.n_candidates = (int)contours.size();
 210     const double amin = p.at("min_outer_area_px2").get<double>(), afrac = p.at("max_outer_area_roi_fraction").get<double>();
 211     const auto wr = p.at("white_outer_area_ratio"), ir = p.at("inner_white_area_ratio");
 212     const double eps = p.at("polygon_epsilon_perimeter_fraction").get<double>(), sidemin = p.at("min_side_length_ratio").get<double>();
 213     int hw = p.at("subpix_half_window").get<int>();
 214     cv::TermCriteria crit(cv::TermCriteria::COUNT | cv::TermCriteria::EPS, p.at("subpix_max_iter").get<int>(), p.at("subpix_eps").get<double>());
 215     struct Pat { cv::Point2d center; std::vector<cv::Point2d> init, refined; double area; };
 216     std::vector<Pat> pats;
 217     for (size_t i = 0; i < contours.size(); ++i) {
 218         int child = hier[i][2];
 219         if (child < 0) continue;
 220         int grand = hier[child][2];
 221         if (grand < 0) continue;
 222         double oa = std::fabs(cv::contourArea(contours[i]));
 223         if (!(oa > amin && oa < afrac * crop.total())) continue;
 224         double ca = std::fabs(cv::contourArea(contours[child])), ia = std::fabs(cv::contourArea(contours[grand]));
 225         if (!(ca / oa > wr[0].get<double>() && ca / oa < wr[1].get<double>() && ia / ca > ir[0].get<double>() && ia / ca < ir[1].get<double>())) continue;
 226         std::vector<cv::Point> quad;
 227         cv::approxPolyDP(contours[i], quad, eps * cv::arcLength(contours[i], true), true);
 228         if (quad.size() != 4 || !cv::isContourConvex(quad)) continue;
 229         auto init = quad_order(quad);
 230         double smin = 1e300, smax = 0;
 231         for (int k = 0; k < 4; ++k) { double s = cv::norm(init[(k + 1) % 4] - init[k]); smin = std::min(smin, s); smax = std::max(smax, s); }
 232         if (smin / smax < sidemin) continue;
 233         std::vector<cv::Point2f> rf(init.begin(), init.end());
 234         cv::cornerSubPix(crop, rf, cv::Size(hw, hw), cv::Size(-1, -1), crit);
 235         Pat pt; pt.area = oa;
 236         for (int k = 0; k < 4; ++k) {
 237             pt.init.emplace_back(init[k].x + t.roi.x, init[k].y + t.roi.y);
 238             pt.refined.emplace_back(rf[k].x + t.roi.x, rf[k].y + t.roi.y);
 239         }
 240         if (!diag_intersection(pt.refined, pt.center)) continue;
 241         pats.push_back(pt);
 242     }
 243     json pj = json::array();
 244     for (auto& q : pats) {
 245         json a = json::array(), b = json::array();
 246         for (int k = 0; k < 4; ++k) { a.push_back({q.init[k].x, q.init[k].y}); b.push_back({q.refined[k].x, q.refined[k].y}); }
 247         pj.push_back({{"center_xy", {q.center.x, q.center.y}}, {"initial_corners_xy", a}, {"refined_corners_xy", b}, {"outer_area_px2", q.area}});
 248     }
 249     r.intermediates["finder_candidates"] = pj;
 250     if (pats.size() != 3) {
 251         r.fail("detection_failed", "expected 3 nested finder patterns, found " + std::to_string(pats.size()));
 252         return r;
 253     }
 254     std::vector<int> idx = {0, 1, 2};
 255     std::sort(idx.begin(), idx.end(), [&](int a, int b) { return pats[a].center.y < pats[b].center.y; });
 256     int A = idx[0], B = idx[1], C = idx[2];
 257     if (pats[B].center.x < pats[A].center.x) std::swap(A, B);
 258     cv::Point2d a = pats[A].center, b = pats[B].center, cc = pats[C].center;
 259     r.intermediates["assignment"] = {{"A_top_left", A}, {"B_top_right", B}, {"C_bottom_left", C}};
 260     if (!(b.x > a.x && cc.y > a.y && std::fabs(cc.x - a.x) < 0.3 * cv::norm(b - a))) {
 261         r.fail("detection_failed", "finder layout check failed (A/B/C arrangement)");
 262         return r;
 263     }
 264     two_point_result(r, a, b, "two finder centres (diagonal intersections of refined outer finder corners), A_top_left -> B_top_right");
 265     r.diagnostics["baseline_length_px"] = cv::norm(b - a);
 266     return r;
 267 }
