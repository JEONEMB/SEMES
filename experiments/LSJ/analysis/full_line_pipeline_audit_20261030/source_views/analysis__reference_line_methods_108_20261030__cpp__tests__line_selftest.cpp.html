   1 // Known-input checks for the line conventions used by line_methods (no image data).
   2 #include "common.hpp"
   3 #include "io_utils.hpp"
   4 
   5 #include <cmath>
   6 #include <cstdio>
   7 
   8 using namespace lm;
   9 
  10 static int failures = 0;
  11 static void check(bool ok, const char* what) {
  12     std::printf("%s %s\n", ok ? "PASS" : "FAIL", what);
  13     if (!ok) ++failures;
  14 }
  15 
  16 int main() {
  17     const double pi = 3.14159265358979323846;
  18     // Endpoint order must not change the angle (no 180 deg flip).
  19     Line a = normalize_line(0, 0, 100, 1), b = normalize_line(100, 1, -100, -1);
  20     check(std::fabs(line_angle_deg(a) - line_angle_deg(b)) < 1e-12, "reversed endpoints give the same angle");
  21     check(std::fabs(line_angle_deg(a) - std::atan(0.01) * 180 / pi) < 1e-12, "slope 0.01 -> atan(0.01) deg");
  22     Line c = normalize_line(0, 0, 100, -1);
  23     check(line_angle_deg(c) < 0, "upward-right line (dy<0) has a negative image angle");
  24     Line v = normalize_line(5, 5, 0, 10);
  25     json vj = line_json(v);
  26     check(std::fabs(line_angle_deg(v) + 90.0) < 1e-12, "vertical line -> -90 deg (range [-90, 90))");
  27     check(vj["slope_dy_dx"].is_null() && vj["intercept_y_px"].is_null(), "vertical line -> slope/intercept null");
  28     auto g = vj["general_form_abc"];
  29     check(std::fabs(g[0].get<double>() * 5 + g[1].get<double>() * 7 + g[2].get<double>()) < 1e-12, "vertical general form passes (5, 7)");
  30     Line h = normalize_line(10, 20, 1, 0);
  31     json hj = line_json(h);
  32     check(std::fabs(hj["slope_dy_dx"].get<double>()) < 1e-15 && std::fabs(hj["intercept_y_px"].get<double>() - 20) < 1e-12, "horizontal y=20");
  33     // Exact points on y = 700 + 0.0047 x must be recovered by the robust fit.
  34     std::vector<cv::Point2d> pts;
  35     for (int x = 24; x <= 3816; x += 2) pts.emplace_back(x, 700 + 0.0047 * x);
  36     pts.emplace_back(2000, 760);  // outlier
  37     FitParams fp; fp.inlier_px = 2.0; fp.min_points = 20;
  38     FitResult f = robust_fit(pts, fp);
  39     json fj = line_json(f.line);
  40     check(f.ok && std::fabs(fj["slope_dy_dx"].get<double>() - 0.0047) < 1e-12 && std::fabs(fj["intercept_y_px"].get<double>() - 700) < 1e-8,
  41           "robust fit recovers slope/intercept with one outlier");
  42     check(f.n_inlier == (int)pts.size() - 1, "outlier trimmed");
  43     // Signed distance sign: +normal is (-dy, dx), i.e. +y for a horizontal line.
  44     check(signed_distance(h, 0, 25) > 0, "signed distance positive below a horizontal line");
  45     std::string abc = "abc";
  46     check(sha256_hex(std::vector<uint8_t>(abc.begin(), abc.end())) == "ba7816bf8f01cfea414140de5dae2223b00361a396177a9cb410ff61f20015ad", "sha256('abc')");
  47     std::printf("%d failure(s)\n", failures);
  48     return failures ? 1 : 0;
  49 }
