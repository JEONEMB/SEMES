cmake_minimum_required(VERSION 3.15)
project(a10_delivered LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(OpenCV 3.4 REQUIRED)
get_filename_component(REPO "${CMAKE_CURRENT_SOURCE_DIR}/../../../.." ABSOLUTE)
set(A0 "${REPO}/experiments/hye/20261102_124309_A0_분할측정고정_계층분할과점측정캐시")
set(COMMON "${REPO}/experiments/hye/00001_common")
include("${REPO}/experiments/selected_modules.cmake")
set(JOINT_DIR "${REPO}/${SEMES_JOINT_DIR}")
set(CONTROL_DIR "${REPO}/${SEMES_CONTROL_DIR}")

# 루트 CMake 와 같은 방식으로 전달 모듈을 세운다 (코드는 그대로 쓴다).
add_library(semes_diagnostics STATIC "${REPO}/src/alignment_diagnostics.cpp")
target_include_directories(semes_diagnostics PUBLIC "${REPO}/include" ${OpenCV_INCLUDE_DIRS})
target_link_libraries(semes_diagnostics PUBLIC ${OpenCV_LIBS})
add_subdirectory("${JOINT_DIR}" joint)
add_subdirectory("${CONTROL_DIR}" control)

add_executable(a10 src/main.cpp)
target_include_directories(a10 PRIVATE "${A0}/실험/src" "${COMMON}/include" "${REPO}/include" ${OpenCV_INCLUDE_DIRS})
target_link_libraries(a10 PRIVATE semes_joint semes_control ${OpenCV_LIBS})
target_compile_definitions(a10 PRIVATE REPO_DIR="${REPO}" EXPERIMENT_DIR="${CMAKE_CURRENT_SOURCE_DIR}/../결과"
  A0_CACHE="${A0}/결과/cache" JOINT_ASSETS="${JOINT_DIR}/assets" CONTROL_ASSETS="${CONTROL_DIR}/assets")
if(MSVC)
 target_compile_options(a10 PRIVATE /utf-8 /EHsc /O2)
 target_compile_definitions(a10 PRIVATE NOMINMAX _CRT_SECURE_NO_WARNINGS _USE_MATH_DEFINES)
endif()
add_custom_command(TARGET a10 POST_BUILD COMMAND ${CMAKE_COMMAND} -E copy_if_different "${OpenCV_DIR}/x64/vc15/bin/opencv_world3415$<$<CONFIG:Debug>:d>.dll" "$<TARGET_FILE_DIR:a10>")
