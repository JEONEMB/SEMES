cmake_minimum_required(VERSION 3.15)
project(subpixel_yt_impact LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(OpenCV 3.4 REQUIRED)
get_filename_component(REPO "${CMAKE_CURRENT_SOURCE_DIR}/../../../.." ABSOLUTE)
set(CONTROL "${REPO}/experiments/hye/00000_final_Y_T control")
# 정합 방법은 1단계 실험, 6후보 학습기는 P12/J204 비교 실험(원본 fold 모델을 1e-6 안으로 재현)의 소스를 그대로 쓴다.
set(STAGE1 "${REPO}/experiments/hye/20261029_122323_부화소정합_합성정답/실험")
set(TRAINER "${REPO}/experiments/hye/20261028_141257_control_P12_vs_J204/실험")
add_executable(yt_impact src/main.cpp "${TRAINER}/src/models.cpp" "${STAGE1}/src/synth.cpp" "${STAGE1}/src/methods.cpp"
 "${CONTROL}/src/tracker.cpp" "${CONTROL}/src/spline.cpp" "${CONTROL}/src/model.cpp")
target_include_directories(yt_impact PRIVATE "${TRAINER}/include" "${STAGE1}/src" "${REPO}/include" "${CONTROL}/include" ${OpenCV_INCLUDE_DIRS})
target_link_libraries(yt_impact PRIVATE ${OpenCV_LIBS} bcrypt)
target_compile_definitions(yt_impact PRIVATE REPO_DIR="${REPO}" EXPERIMENT_DIR="${CMAKE_CURRENT_SOURCE_DIR}/../결과")
if(MSVC)
 target_compile_options(yt_impact PRIVATE /utf-8 /EHsc)
 target_compile_definitions(yt_impact PRIVATE NOMINMAX _CRT_SECURE_NO_WARNINGS)
endif()
# 보조: 20261029 혼합 234장 날짜 간 참고 평가 (results/models의 전체 학습 모델 사용)
add_executable(yt_transfer_1029 src/transfer_1029.cpp "${TRAINER}/src/models.cpp" "${STAGE1}/src/synth.cpp" "${STAGE1}/src/methods.cpp"
 "${CONTROL}/src/tracker.cpp" "${CONTROL}/src/spline.cpp")
target_include_directories(yt_transfer_1029 PRIVATE "${TRAINER}/include" "${STAGE1}/src" "${REPO}/include" "${CONTROL}/include" ${OpenCV_INCLUDE_DIRS})
target_link_libraries(yt_transfer_1029 PRIVATE ${OpenCV_LIBS} bcrypt)
target_compile_definitions(yt_transfer_1029 PRIVATE REPO_DIR="${REPO}" EXPERIMENT_DIR="${CMAKE_CURRENT_SOURCE_DIR}/../결과")
if(MSVC)
 target_compile_options(yt_transfer_1029 PRIVATE /utf-8 /EHsc)
 target_compile_definitions(yt_transfer_1029 PRIVATE NOMINMAX _CRT_SECURE_NO_WARNINGS)
endif()
add_custom_command(TARGET yt_impact POST_BUILD COMMAND ${CMAKE_COMMAND} -E copy_if_different "${OpenCV_DIR}/x64/vc15/bin/opencv_world3415$<$<CONFIG:Debug>:d>.dll" "$<TARGET_FILE_DIR:yt_impact>")
