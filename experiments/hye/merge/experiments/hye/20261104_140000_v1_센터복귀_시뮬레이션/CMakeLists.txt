cmake_minimum_required(VERSION 3.15)
project(return_simulation LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
find_package(OpenCV 3.4 REQUIRED)
add_executable(return_sim src/main.cpp)
target_include_directories(return_sim PRIVATE ${OpenCV_INCLUDE_DIRS})
target_link_libraries(return_sim PRIVATE ${OpenCV_LIBS})
target_compile_options(return_sim PRIVATE /utf-8 /O2 /EHsc)
target_compile_definitions(return_sim PRIVATE NOMINMAX)
