cmake_minimum_required(VERSION 3.15)
project(origin_ablation LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
find_package(OpenCV 3.4 REQUIRED)
get_filename_component(REPO "${CMAKE_CURRENT_SOURCE_DIR}/../../.." ABSOLUTE)
set(PROJECT_SOURCE_DIR "${REPO}")
add_library(semes_diagnostics STATIC "${REPO}/src/alignment_diagnostics.cpp")
target_include_directories(semes_diagnostics PUBLIC "${REPO}/include" ${OpenCV_INCLUDE_DIRS})
target_link_libraries(semes_diagnostics PUBLIC ${OpenCV_LIBS})
set(J "${REPO}/experiments/hye/20261104_1435_v2_joint_강등")
set(C "${REPO}/experiments/hye/20261104_1435_v2_control_강등")
add_subdirectory("${J}" joint)
add_subdirectory("${C}" control)
add_executable(origin_eval src/main.cpp)
target_include_directories(origin_eval PRIVATE "${J}/include" "${C}/include")
target_link_libraries(origin_eval PRIVATE semes_joint semes_control)
target_compile_options(origin_eval PRIVATE /utf-8 /EHsc /O2)
target_compile_definitions(origin_eval PRIVATE NOMINMAX)
