cmake_minimum_required(VERSION 3.15)
project(a16_noise_zero LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(OpenCV 3.4 REQUIRED)
get_filename_component(REPO "${CMAKE_CURRENT_SOURCE_DIR}/../../../.." ABSOLUTE)
set(A0 "${REPO}/experiments/hye/20261102_124309_A0_분할측정고정_계층분할과점측정캐시")
set(COMMON "${REPO}/experiments/hye/00001_common")
add_executable(a16 src/main.cpp)
target_include_directories(a16 PRIVATE "${A0}/실험/src" "${COMMON}/include" ${OpenCV_INCLUDE_DIRS})
target_link_libraries(a16 PRIVATE ${OpenCV_LIBS})
target_compile_definitions(a16 PRIVATE REPO_DIR="${REPO}" EXPERIMENT_DIR="${CMAKE_CURRENT_SOURCE_DIR}/../결과" A0_CACHE="${A0}/결과/cache")
if(MSVC)
 target_compile_options(a16 PRIVATE /utf-8 /EHsc /O2 /openmp)
 target_compile_definitions(a16 PRIVATE NOMINMAX _CRT_SECURE_NO_WARNINGS _USE_MATH_DEFINES)
endif()
add_custom_command(TARGET a16 POST_BUILD COMMAND ${CMAKE_COMMAND} -E copy_if_different "${OpenCV_DIR}/x64/vc15/bin/opencv_world3415$<$<CONFIG:Debug>:d>.dll" "$<TARGET_FILE_DIR:a16>")
