cmake_minimum_required(VERSION 3.15)
project(pc_bench LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(OpenCV REQUIRED)
# 00000_final_Y_T_semes 모듈(semes_joint)을 그대로 빌드해 나사 측정 클래스를 쓴다.
get_filename_component(REPO "${CMAKE_CURRENT_SOURCE_DIR}/../../../../.." ABSOLUTE)
set(MODULE "${REPO}/experiments/hye/00000_final_Y_T_semes")
set(PROJECT_SOURCE_DIR "${REPO}")
add_library(semes_diagnostics STATIC "${REPO}/src/alignment_diagnostics.cpp")
target_include_directories(semes_diagnostics PUBLIC "${REPO}/include" ${OpenCV_INCLUDE_DIRS})
target_link_libraries(semes_diagnostics PUBLIC ${OpenCV_LIBS})
if(MSVC)
    target_compile_options(semes_diagnostics PRIVATE /utf-8 /EHsc)
    target_compile_definitions(semes_diagnostics PRIVATE NOMINMAX)
endif()
add_subdirectory("${MODULE}" joint)
add_executable(pc_bench pc_bench.cpp)
target_include_directories(pc_bench PRIVATE "${MODULE}/include")
target_link_libraries(pc_bench PRIVATE semes_joint)
target_compile_definitions(pc_bench PRIVATE NOMINMAX _CRT_SECURE_NO_WARNINGS)
if(MSVC)
    target_compile_options(pc_bench PRIVATE /utf-8 /EHsc)
    add_custom_command(TARGET pc_bench POST_BUILD COMMAND ${CMAKE_COMMAND} -E copy_if_different
        "${OpenCV_DIR}/x64/vc15/bin/opencv_world3415.dll" "$<TARGET_FILE_DIR:pc_bench>")
endif()
