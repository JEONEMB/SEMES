cmake_minimum_required(VERSION 3.15)
project(icgn_bench LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(OpenCV REQUIRED)
# 00000_final_Y_T_semes 모듈(semes_joint)과 KSH 정밀 맞춤 복사본(ksh/, origin/KSH 429b5533 에서 내용 그대로)을 함께 빌드한다.
get_filename_component(REPO "${CMAKE_CURRENT_SOURCE_DIR}/../../../../.." ABSOLUTE)
set(MODULE "${REPO}/experiments/hye/00000_final_Y_T_semes")
set(PROJECT_SOURCE_DIR "${REPO}")
add_library(semes_diagnostics STATIC "${REPO}/src/alignment_diagnostics.cpp")
target_include_directories(semes_diagnostics PUBLIC "${REPO}/include" ${OpenCV_INCLUDE_DIRS})
target_link_libraries(semes_diagnostics PUBLIC ${OpenCV_LIBS})
if(MSVC)
    target_compile_options(semes_diagnostics PRIVATE /utf-8 /EHsc)
    target_compile_definitions(semes_diagnostics PRIVATE NOMINMAX)
endif()
add_subdirectory("${MODULE}" joint)
add_library(ksh_fine STATIC ksh/src/fine_align.cpp ksh/src/interp_kernel.cpp ksh/src/image_util.cpp)
target_include_directories(ksh_fine PUBLIC ksh/include ${OpenCV_INCLUDE_DIRS})
target_link_libraries(ksh_fine PUBLIC ${OpenCV_LIBS})
add_executable(icgn_bench icgn_bench.cpp)
target_include_directories(icgn_bench PRIVATE "${MODULE}/include")
target_link_libraries(icgn_bench PRIVATE semes_joint ksh_fine)
target_compile_definitions(icgn_bench PRIVATE NOMINMAX _CRT_SECURE_NO_WARNINGS)
if(MSVC)
    target_compile_options(ksh_fine PRIVATE /utf-8 /EHsc)
    target_compile_definitions(ksh_fine PRIVATE NOMINMAX _CRT_SECURE_NO_WARNINGS _USE_MATH_DEFINES)
    target_compile_options(icgn_bench PRIVATE /utf-8 /EHsc)
    add_custom_command(TARGET icgn_bench POST_BUILD COMMAND ${CMAKE_COMMAND} -E copy_if_different
        "${OpenCV_DIR}/x64/vc15/bin/opencv_world3415.dll" "$<TARGET_FILE_DIR:icgn_bench>")
endif()
