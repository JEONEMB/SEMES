cmake_minimum_required(VERSION 3.15)
project(subpixel_synthetic_benchmark LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(OpenCV 3.4 REQUIRED)
get_filename_component(REPO "${CMAKE_CURRENT_SOURCE_DIR}/../../../.." ABSOLUTE)
set(CONTROL "${REPO}/experiments/hye/20261028_1327_final_Y_T control")
add_executable(subpixel_synthetic src/main.cpp src/synth.cpp src/methods.cpp "${CONTROL}/src/tracker.cpp" "${CONTROL}/src/spline.cpp")
target_include_directories(subpixel_synthetic PRIVATE src "${REPO}/include" "${CONTROL}/include" ${OpenCV_INCLUDE_DIRS})
target_link_libraries(subpixel_synthetic PRIVATE ${OpenCV_LIBS} bcrypt)
target_compile_definitions(subpixel_synthetic PRIVATE REPO_DIR="${REPO}" EXPERIMENT_DIR="${CMAKE_CURRENT_SOURCE_DIR}/../결과")
if(MSVC)
 target_compile_options(subpixel_synthetic PRIVATE /utf-8 /EHsc)
 target_compile_definitions(subpixel_synthetic PRIVATE NOMINMAX _CRT_SECURE_NO_WARNINGS)
endif()
add_custom_command(TARGET subpixel_synthetic POST_BUILD COMMAND ${CMAKE_COMMAND} -E copy_if_different "${OpenCV_DIR}/x64/vc15/bin/opencv_world3415$<$<CONFIG:Debug>:d>.dll" "$<TARGET_FILE_DIR:subpixel_synthetic>")
