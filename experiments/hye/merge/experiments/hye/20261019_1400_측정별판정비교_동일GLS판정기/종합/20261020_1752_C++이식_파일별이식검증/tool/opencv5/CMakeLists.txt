cmake_minimum_required(VERSION 3.26)
project(SEMES_Observer5 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
get_filename_component(SEMES_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../../.." ABSOLUTE)
set(SEMES_OPENCV5_DIR "${SEMES_ROOT}/third_party/opencv500-build" CACHE PATH "Pinned native OpenCV5 build")
set(OpenCV_DIR "${SEMES_OPENCV5_DIR}")
find_package(OpenCV 5.0.0 EXACT REQUIRED COMPONENTS core imgproc imgcodecs video features)
add_custom_target(observer5_runtime ALL COMMAND ${CMAKE_COMMAND} -E copy_directory_if_different "${SEMES_OPENCV5_DIR}/bin/Release" "${CMAKE_RUNTIME_OUTPUT_DIRECTORY}/$<CONFIG>")
if(MSVC)
  add_compile_options(/utf-8 /EHsc)
  add_compile_definitions(NOMINMAX _CRT_SECURE_NO_WARNINGS)
endif()
set(ZSRC adler32 compress crc32 deflate inflate infback inftrees inffast trees uncompr zutil)
foreach(name IN LISTS ZSRC)
  list(APPEND ZFILES "${SEMES_ROOT}/third_party/zlib/${name}.c")
endforeach()
enable_testing()
add_executable(motion_checks "${SEMES_ROOT}/tests/motion_checks.cpp")
target_link_libraries(motion_checks PRIVATE observer5)
add_dependencies(motion_checks observer5_runtime)
add_test(NAME modern_acknowledged_motion_checks COMMAND motion_checks "${CMAKE_BINARY_DIR}/test-work/motion" --unique)
add_executable(historical_measure_qr "${SEMES_ROOT}/experiments/analysis_v1_20261012_3QR측정유지_회귀C++이식/src/measure_qr.cpp")
target_link_libraries(historical_measure_qr PRIVATE observer5)
add_dependencies(historical_measure_qr observer5_runtime)
add_library(observer_zlib STATIC ${ZFILES})
target_include_directories(observer_zlib PUBLIC "${SEMES_ROOT}/third_party/zlib")
add_library(observer_io "${SEMES_ROOT}/src/io.cpp")
target_include_directories(observer_io PUBLIC "${SEMES_ROOT}/include")
target_link_libraries(observer_io PUBLIC observer_zlib bcrypt)
add_library(observer5 "${SEMES_ROOT}/src/observer_support.cpp" "${SEMES_ROOT}/src/direct_observer.cpp" "${SEMES_ROOT}/src/line_observer.cpp" "${SEMES_ROOT}/src/ecc5_bridge.cpp" "${SEMES_ROOT}/src/motion.cpp")
target_include_directories(observer5 PUBLIC "${SEMES_ROOT}/include" ${OpenCV_INCLUDE_DIRS})
target_link_libraries(observer5 PUBLIC observer_io ${OpenCV_LIBS})
add_executable(historical_v2_homography "${SEMES_ROOT}/experiments/analysis_v2_20261013_특징분할유지_통계C++이식/src/extract_homography.cpp")
target_link_libraries(historical_v2_homography PRIVATE observer5)
add_dependencies(historical_v2_homography observer5_runtime)
add_executable(historical_v1_track "${SEMES_ROOT}/experiments/analysis_v1_20261013_고정점분할유지_선형계산C++이식/src/track_all.cpp")
target_link_libraries(historical_v1_track PRIVATE observer5)
add_dependencies(historical_v1_track observer5_runtime)
foreach(app direct_extract image_extract image_session calibrate calibrate_full motion_session prepare_features feedback_step feedback_image_session forward_gls predict_observation tolerance_metrics fit_observation evaluate_static generalization_features generalization_data paired_ridge paired_screen)
  add_executable(${app} "${SEMES_ROOT}/tool/${app}.cpp")
  target_link_libraries(${app} PRIVATE observer5)
  add_dependencies(${app} observer5_runtime)
endforeach()
