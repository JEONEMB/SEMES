option(SEMES_ALLOW_UNVALIDATED_PLATFORM "Allow non-Windows builds for compatibility audits only" OFF)
if(NOT WIN32 AND NOT SEMES_ALLOW_UNVALIDATED_PLATFORM)
    message(FATAL_ERROR "final66 reference observations were measured on Windows x64/MSVC v141. Non-Windows builds require SEMES_ALLOW_UNVALIDATED_PLATFORM=ON for audits; image-coordinate parity is not certified.")
endif()
option(SEMES_ALLOW_UNVALIDATED_OPENCV "Allow other OpenCV versions for compatibility audits only" OFF)
if(NOT OpenCV_VERSION VERSION_EQUAL "3.4.15")
    if(NOT SEMES_ALLOW_UNVALIDATED_OPENCV)
        message(FATAL_ERROR "final66 requires OpenCV 3.4.15. Other versions changed image observations in the local audit. For diagnosis only, explicitly set SEMES_ALLOW_UNVALIDATED_OPENCV=ON.")
    endif()
    message(WARNING "Unvalidated OpenCV version: ${OpenCV_VERSION}; do not reuse the reference accuracy claims")
endif()
add_library(semes_joint STATIC src/joint.cpp src/estimator.cpp src/model.cpp
    src/observer.cpp src/line_profile.cpp src/pin_observer.cpp third_party/ecc_compat.cpp)
target_include_directories(semes_joint PUBLIC "${PROJECT_SOURCE_DIR}/include" ${OpenCV_INCLUDE_DIRS} PRIVATE include)
target_link_libraries(semes_joint PUBLIC semes_diagnostics ${OpenCV_LIBS})
target_compile_features(semes_joint PUBLIC cxx_std_17)
if(WIN32)
    target_link_libraries(semes_joint PUBLIC bcrypt)
endif()
if(MSVC)
    target_compile_options(semes_joint PRIVATE /utf-8 /EHsc)
    target_compile_definitions(semes_joint PRIVATE NOMINMAX _CRT_SECURE_NO_WARNINGS)
endif()
add_executable(final66_tool tool/final66_tool.cpp)
target_include_directories(final66_tool PRIVATE include)
target_link_libraries(final66_tool PRIVATE semes_joint)
add_executable(final66_checks tests/final66_checks.cpp "${PROJECT_SOURCE_DIR}/src/alignment.cpp")
target_include_directories(final66_checks PRIVATE include)
target_link_libraries(final66_checks PRIVATE semes_joint)
if(MSVC)
    foreach(app final66_tool final66_checks)
        target_compile_options(${app} PRIVATE /utf-8 /EHsc)
        target_compile_definitions(${app} PRIVATE NOMINMAX)
        add_custom_command(TARGET ${app} POST_BUILD COMMAND ${CMAKE_COMMAND} -E copy_if_different
            "${OpenCV_DIR}/x64/vc15/bin/opencv_world3415$<$<CONFIG:Debug>:d>.dll" "$<TARGET_FILE_DIR:${app}>")
    endforeach()
endif()
add_test(NAME final66_contract COMMAND final66_checks "${CMAKE_CURRENT_SOURCE_DIR}")
