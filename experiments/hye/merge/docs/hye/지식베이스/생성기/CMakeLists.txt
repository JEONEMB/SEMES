cmake_minimum_required(VERSION 3.15)
project(kb_generator LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
get_filename_component(KB "${CMAKE_CURRENT_SOURCE_DIR}/.." ABSOLUTE)
get_filename_component(LIT "${KB}/../문헌연구" ABSOLUTE)
get_filename_component(REPO "${KB}/../../.." ABSOLUTE)
# 기본 대상 캔버스. 실행 인자로 다른 경로를 줄 수 있다.
if(NOT DEFINED SEMES_CANVAS)
  set(SEMES_CANVAS "C:/Users/SSAFY/Desktop/메모/세메스_정리.canvas")
endif()
add_executable(kbgen src/main.cpp)
target_compile_definitions(kbgen PRIVATE KB_DIR="${KB}" LIT_DIR="${LIT}" REPO_DIR="${REPO}" CANVAS_PATH="${SEMES_CANVAS}")
if(MSVC)
  target_compile_options(kbgen PRIVATE /utf-8 /EHsc /O2)
  target_compile_definitions(kbgen PRIVATE NOMINMAX _CRT_SECURE_NO_WARNINGS)
endif()
