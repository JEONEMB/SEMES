cmake_minimum_required(VERSION 3.15)
project(control_lk_measurement LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(OpenCV 3.4 REQUIRED)
get_filename_component(REPO "${CMAKE_CURRENT_SOURCE_DIR}/../../../.." ABSOLUTE)
set(CONTROL "${REPO}/experiments/hye/00000_final_Y_T control")
# 학습식(6절편·2차 응답·축소 0.05)은 P12/J204 비교 실험의 train_yort를 그대로 쓴다
set(STUDY "${REPO}/experiments/hye/20261028_141257_control_P12_vs_J204/실험 방법")
add_executable(compare_lk src/main.cpp "${STUDY}/src/models.cpp" "${CONTROL}/src/tracker.cpp" "${CONTROL}/src/spline.cpp" "${CONTROL}/src/model.cpp")
target_include_directories(compare_lk PRIVATE "${STUDY}/include" "${REPO}/include" "${CONTROL}/include" ${OpenCV_INCLUDE_DIRS})
target_link_libraries(compare_lk PRIVATE ${OpenCV_LIBS} bcrypt)
target_compile_definitions(compare_lk PRIVATE REPO_DIR="${REPO}" EXPERIMENT_DIR="${CMAKE_CURRENT_SOURCE_DIR}/../결과 정리")
if(MSVC)
 target_compile_options(compare_lk PRIVATE /utf-8 /EHsc)
 target_compile_definitions(compare_lk PRIVATE NOMINMAX _CRT_SECURE_NO_WARNINGS)
endif()
add_custom_command(TARGET compare_lk POST_BUILD COMMAND ${CMAKE_COMMAND} -E copy_if_different "${OpenCV_DIR}/x64/vc15/bin/opencv_world3415$<$<CONFIG:Debug>:d>.dll" "$<TARGET_FILE_DIR:compare_lk>")
