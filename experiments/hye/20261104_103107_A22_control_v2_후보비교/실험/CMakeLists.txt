cmake_minimum_required(VERSION 3.15)
project(a22 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(OpenCV REQUIRED)
get_filename_component(REPO "${CMAKE_CURRENT_SOURCE_DIR}/../../../.." ABSOLUTE)
set(A0 "${REPO}/experiments/hye/20261102_124309_A0_분할측정고정_계층분할과점측정캐시")
add_executable(a22 src/main.cpp)
target_include_directories(a22 PRIVATE "${A0}/실험/src" "${REPO}/experiments/hye/00001_common/include" ${OpenCV_INCLUDE_DIRS})
target_link_libraries(a22 PRIVATE ${OpenCV_LIBS})
target_compile_definitions(a22 PRIVATE REPO_DIR="${REPO}" EXPERIMENT_DIR="${CMAKE_CURRENT_SOURCE_DIR}/../결과" A0_CACHE="${A0}/결과/cache")
if(MSVC)
 target_compile_options(a22 PRIVATE /utf-8 /EHsc /O2)
 target_compile_definitions(a22 PRIVATE NOMINMAX _CRT_SECURE_NO_WARNINGS _USE_MATH_DEFINES)
endif()
