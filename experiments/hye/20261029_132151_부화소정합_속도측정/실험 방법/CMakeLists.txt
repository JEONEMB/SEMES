cmake_minimum_required(VERSION 3.15)
project(subpixel_speed LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(OpenCV 3.4 REQUIRED)
get_filename_component(REPO "${CMAKE_CURRENT_SOURCE_DIR}/../../../.." ABSOLUTE)
set(CONTROL "${REPO}/experiments/hye/00000_final_Y_T control")
set(STAGE1 "${REPO}/experiments/hye/20261029_122323_부화소정합_합성정답/실험 방법")
set(TRAINER "${REPO}/experiments/hye/20261028_141257_control_P12_vs_J204/실험 방법")
add_executable(subpixel_speed src/main.cpp "${TRAINER}/src/models.cpp" "${STAGE1}/src/synth.cpp" "${STAGE1}/src/methods.cpp"
 "${CONTROL}/src/tracker.cpp" "${CONTROL}/src/spline.cpp")
target_include_directories(subpixel_speed PRIVATE "${TRAINER}/include" "${STAGE1}/src" "${REPO}/include" "${CONTROL}/include" ${OpenCV_INCLUDE_DIRS})
target_link_libraries(subpixel_speed PRIVATE ${OpenCV_LIBS} bcrypt)
target_compile_definitions(subpixel_speed PRIVATE REPO_DIR="${REPO}" EXPERIMENT_DIR="${CMAKE_CURRENT_SOURCE_DIR}/../결과 정리")
if(MSVC)
 target_compile_options(subpixel_speed PRIVATE /utf-8 /EHsc)
 target_compile_definitions(subpixel_speed PRIVATE NOMINMAX _CRT_SECURE_NO_WARNINGS)
endif()
add_custom_command(TARGET subpixel_speed POST_BUILD COMMAND ${CMAKE_COMMAND} -E copy_if_different "${OpenCV_DIR}/x64/vc15/bin/opencv_world3415$<$<CONFIG:Debug>:d>.dll" "$<TARGET_FILE_DIR:subpixel_speed>")
