cmake_minimum_required(VERSION 3.15)
project(center_zero LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(OpenCV REQUIRED)
get_filename_component(REPO "${CMAKE_CURRENT_SOURCE_DIR}/../../../.." ABSOLUTE)
# 측정 코드는 A19 의 실험/src 를 그대로 쓴다 (복사본을 만들지 않는다)
set(A19 "${REPO}/experiments/hye/20261104_093558_A19_QR선측정_안정성점검/실험")
add_executable(center_zero src/main.cpp "${A19}/src/observer.cpp" "${A19}/src/line_profile.cpp" "${A19}/src/ecc_compat.cpp")
target_include_directories(center_zero PRIVATE "${A19}/include" "${REPO}/include" "${REPO}/experiments/hye/00001_common/include" ${OpenCV_INCLUDE_DIRS})
target_link_libraries(center_zero PRIVATE ${OpenCV_LIBS})
if(WIN32)
 target_link_libraries(center_zero PRIVATE bcrypt)   # observer.cpp 의 sha256 (BCrypt)
endif()
target_compile_definitions(center_zero PRIVATE REPO_DIR="${REPO}" EXPERIMENT_DIR="${CMAKE_CURRENT_SOURCE_DIR}/../결과")
if(MSVC)
 target_compile_options(center_zero PRIVATE /utf-8 /EHsc /O2)
 target_compile_definitions(center_zero PRIVATE NOMINMAX _CRT_SECURE_NO_WARNINGS _USE_MATH_DEFINES)
endif()
