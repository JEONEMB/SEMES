cmake_minimum_required(VERSION 3.15)
project(perturb_test LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(OpenCV REQUIRED)
# 모듈 코드는 세 모듈(ssafy, ssafy_2, semes)이 같다. ssafy 모듈 소스를 빌드하고 assets 는 실행할 때 고른다.
get_filename_component(REPO "${CMAKE_CURRENT_SOURCE_DIR}/../../../.." ABSOLUTE)
set(MODULE "${REPO}/experiments/hye/00000_final_Y_T_ssafy")
set(PROJECT_SOURCE_DIR "${REPO}")
add_library(semes_diagnostics STATIC "${REPO}/src/alignment_diagnostics.cpp")
target_include_directories(semes_diagnostics PUBLIC "${REPO}/include" ${OpenCV_INCLUDE_DIRS})
target_link_libraries(semes_diagnostics PUBLIC ${OpenCV_LIBS})
if(MSVC)
    target_compile_options(semes_diagnostics PRIVATE /utf-8 /EHsc)
    target_compile_definitions(semes_diagnostics PRIVATE NOMINMAX)
endif()
add_subdirectory("${MODULE}" joint)
add_executable(perturb_test perturb_test.cpp)
target_include_directories(perturb_test PRIVATE "${MODULE}/include")
target_link_libraries(perturb_test PRIVATE semes_joint)
target_compile_definitions(perturb_test PRIVATE NOMINMAX _USE_MATH_DEFINES legacy_control=final00_tracking)
if(MSVC)
    target_compile_options(perturb_test PRIVATE /utf-8 /EHsc /openmp)
endif()
