cmake_minimum_required(VERSION 3.15)
project(fivefold_eval LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
find_package(OpenCV 3.4 REQUIRED)
get_filename_component(REPO "${CMAKE_CURRENT_SOURCE_DIR}/../../.." ABSOLUTE)
set(PROJECT_SOURCE_DIR "${REPO}")
add_library(semes_diagnostics STATIC "${REPO}/src/alignment_diagnostics.cpp")
target_include_directories(semes_diagnostics PUBLIC "${REPO}/include" ${OpenCV_INCLUDE_DIRS})
target_link_libraries(semes_diagnostics PUBLIC ${OpenCV_LIBS})
set(J "${REPO}/experiments/hye/20261104_1435_v2_joint_강등")
set(C "${REPO}/experiments/hye/20261104_1435_v2_control_강등")
add_subdirectory("${J}" joint)
add_subdirectory("${C}" control)
add_executable(eval src/eval.cpp)
target_include_directories(eval PRIVATE "${J}/include" "${C}/include")
target_link_libraries(eval PRIVATE semes_joint semes_control)
target_compile_options(eval PRIVATE /utf-8 /EHsc /O2)
target_compile_definitions(eval PRIVATE NOMINMAX)
add_executable(eval_frozen src/frozen.cpp)
target_include_directories(eval_frozen PRIVATE "${J}/include" "${C}/include")
target_link_libraries(eval_frozen PRIVATE semes_joint semes_control)
target_compile_options(eval_frozen PRIVATE /utf-8 /EHsc /O2)
target_compile_definitions(eval_frozen PRIVATE NOMINMAX)
get_filename_component(LSJ "${REPO}/../S15P31A507-LSJ" ABSOLUTE)
add_library(lsj_core STATIC "${LSJ}/src/theta_alignment.cpp" "${LSJ}/src/y_alignment.cpp" "${LSJ}/src/sequential_alignment.cpp" "${LSJ}/src/path_validation.cpp")
target_include_directories(lsj_core PUBLIC "${LSJ}/include" ${OpenCV_INCLUDE_DIRS})
target_link_libraries(lsj_core PUBLIC ${OpenCV_LIBS})
target_compile_options(lsj_core PRIVATE /utf-8 /EHsc /O2)
target_compile_definitions(lsj_core PRIVATE NOMINMAX)
add_executable(eval_lsj src/lsj.cpp)
target_link_libraries(eval_lsj PRIVATE lsj_core)
target_compile_options(eval_lsj PRIVATE /utf-8 /EHsc /O2)
target_compile_definitions(eval_lsj PRIVATE NOMINMAX)
set(NEWLSJ "${LSJ}/experiments/LSJ/algorithm/theta_huber_20261101")
add_library(lsj_new STATIC "${NEWLSJ}/src/edge_points.cpp" "${NEWLSJ}/src/huber_line.cpp" "${NEWLSJ}/src/theta_decoder.cpp" "${NEWLSJ}/src/theta_pipeline.cpp" "${NEWLSJ}/src/utf8_io.cpp")
target_include_directories(lsj_new PUBLIC "${NEWLSJ}/include" ${OpenCV_INCLUDE_DIRS})
target_link_libraries(lsj_new PUBLIC ${OpenCV_LIBS})
target_compile_options(lsj_new PRIVATE /utf-8 /EHsc /O2)
add_executable(eval_lsj_new src/lsj_new.cpp)
target_link_libraries(eval_lsj_new PRIVATE lsj_new)
target_compile_options(eval_lsj_new PRIVATE /utf-8 /EHsc /O2)
