cmake_minimum_required(VERSION 3.15)
project(public_api_benchmark LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
find_package(OpenCV 3.4 REQUIRED)
get_filename_component(REPO "${CMAKE_CURRENT_SOURCE_DIR}/../../../.." ABSOLUTE)
set(PROJECT_SOURCE_DIR "${REPO}")
add_library(semes_diagnostics STATIC "${REPO}/src/alignment_diagnostics.cpp")
target_include_directories(semes_diagnostics PUBLIC "${REPO}/include" ${OpenCV_INCLUDE_DIRS})
target_link_libraries(semes_diagnostics PUBLIC ${OpenCV_LIBS})
if(EVAL_VERSION STREQUAL "2")
 set(J "${REPO}/experiments/hye/v2_Y_T_joint")
 set(C "${REPO}/experiments/hye/v2_Y_T control")
else()
 set(J "${REPO}/experiments/hye/00000_final_Y_T_joint")
 set(C "${REPO}/experiments/hye/00000_final_Y_T control")
endif()
add_subdirectory("${J}" joint)
add_subdirectory("${C}" control)
add_executable(bench ../src/bench.cpp)
target_link_libraries(bench PRIVATE semes_joint semes_control)
target_compile_options(bench PRIVATE /utf-8 /EHsc /O2)
target_compile_definitions(bench PRIVATE NOMINMAX)
