cmake_minimum_required(VERSION 3.15)
project(a19 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(OpenCV REQUIRED)
find_package(Threads REQUIRED)
get_filename_component(REPO "${CMAKE_CURRENT_SOURCE_DIR}/../../../.." ABSOLUTE)
add_executable(a19 src/main.cpp src/observer.cpp src/line_profile.cpp src/ecc_compat.cpp)
target_include_directories(a19 PRIVATE include "${REPO}/include" ${OpenCV_INCLUDE_DIRS})
target_link_libraries(a19 PRIVATE ${OpenCV_LIBS} Threads::Threads)
target_compile_definitions(a19 PRIVATE REPO_DIR="${REPO}" EXPERIMENT_DIR="${CMAKE_CURRENT_SOURCE_DIR}/../결과")
if(MSVC)
 target_compile_options(a19 PRIVATE /utf-8 /EHsc /O2)
 target_compile_definitions(a19 PRIVATE NOMINMAX _CRT_SECURE_NO_WARNINGS _USE_MATH_DEFINES)
endif()
