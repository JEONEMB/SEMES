cmake_minimum_required(VERSION 3.15)
project(detector_synthetic_benchmark LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(OpenCV 3.4 REQUIRED)
get_filename_component(REPO "${CMAKE_CURRENT_SOURCE_DIR}/../../../.." ABSOLUTE)
set(CONTROL "${REPO}/experiments/hye/20261028_1327_final_Y_T control")
# 합성 조건 생성기와 추적 방법은 1단계 실험의 소스를 그대로 쓴다 (같은 조건·같은 LK 설정).
set(STAGE1 "${REPO}/experiments/hye/20261029_122323_부화소정합_합성정답/실험")
add_executable(detector_synthetic src/main.cpp src/detectors.cpp "${STAGE1}/src/synth.cpp" "${STAGE1}/src/methods.cpp" "${CONTROL}/src/tracker.cpp" "${CONTROL}/src/spline.cpp")
target_include_directories(detector_synthetic PRIVATE src "${STAGE1}/src" "${REPO}/include" "${CONTROL}/include" ${OpenCV_INCLUDE_DIRS})
target_link_libraries(detector_synthetic PRIVATE ${OpenCV_LIBS} bcrypt)
target_compile_definitions(detector_synthetic PRIVATE REPO_DIR="${REPO}" EXPERIMENT_DIR="${CMAKE_CURRENT_SOURCE_DIR}/../결과")
if(MSVC)
 target_compile_options(detector_synthetic PRIVATE /utf-8 /EHsc)
 target_compile_definitions(detector_synthetic PRIVATE NOMINMAX _CRT_SECURE_NO_WARNINGS)
endif()
add_custom_command(TARGET detector_synthetic POST_BUILD COMMAND ${CMAKE_COMMAND} -E copy_if_different "${OpenCV_DIR}/x64/vc15/bin/opencv_world3415$<$<CONFIG:Debug>:d>.dll" "$<TARGET_FILE_DIR:detector_synthetic>")
