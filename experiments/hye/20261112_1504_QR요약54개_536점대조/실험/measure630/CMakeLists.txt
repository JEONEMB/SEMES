cmake_minimum_required(VERSION 3.15)
project(measure630 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(OpenCV 3.4 REQUIRED)
get_filename_component(REPO "${CMAKE_CURRENT_SOURCE_DIR}/../../../../.." ABSOLUTE)
set(CONTROL "${REPO}/experiments/hye/00000_final_Y_T control")
# 측정은 채택 control 모듈의 tracker 소스를 그대로 링크한다 (A0 캐시와 같은 측정기).
add_executable(measure630 src/main.cpp "${CONTROL}/src/tracker.cpp" "${CONTROL}/src/spline.cpp")
target_include_directories(measure630 PRIVATE "${REPO}/include" "${CONTROL}/include" ${OpenCV_INCLUDE_DIRS})
target_link_libraries(measure630 PRIVATE ${OpenCV_LIBS} bcrypt)
if(MSVC)
 target_compile_options(measure630 PRIVATE /utf-8 /EHsc /O2 /openmp)
 target_compile_definitions(measure630 PRIVATE NOMINMAX _CRT_SECURE_NO_WARNINGS _USE_MATH_DEFINES)
endif()
add_custom_command(TARGET measure630 POST_BUILD COMMAND ${CMAKE_COMMAND} -E copy_if_different "${OpenCV_DIR}/x64/vc15/bin/opencv_world3415$<$<CONFIG:Debug>:d>.dll" "$<TARGET_FILE_DIR:measure630>")
