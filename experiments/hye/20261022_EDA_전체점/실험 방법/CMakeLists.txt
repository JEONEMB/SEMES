cmake_minimum_required(VERSION 3.15)
project(export_points LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
find_package(OpenCV 3.4 REQUIRED)
set(SRC "${CMAKE_CURRENT_LIST_DIR}/../hw-analysis/tool/hw/eda/src")
add_executable(export_points src/export.cpp "${SRC}/core.cpp" "${SRC}/measure.cpp" "${SRC}/measurements.cpp")
target_include_directories(export_points PRIVATE "${SRC}")
target_link_libraries(export_points PRIVATE ${OpenCV_LIBS})
target_compile_options(export_points PRIVATE /utf-8 /W3)
target_compile_definitions(export_points PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN)
add_custom_command(TARGET export_points POST_BUILD COMMAND ${CMAKE_COMMAND} -E copy_if_different "${OpenCV_DIR}/x64/vc15/bin/opencv_world3415.dll" "$<TARGET_FILE_DIR:export_points>")
