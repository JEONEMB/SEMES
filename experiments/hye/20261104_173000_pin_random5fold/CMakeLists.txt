cmake_minimum_required(VERSION 3.15)
project(pin_random5fold LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
find_package(OpenCV 3.4 REQUIRED)
get_filename_component(REPO "${CMAKE_CURRENT_SOURCE_DIR}/../../.." ABSOLUTE)
set(PROJECT_SOURCE_DIR "${REPO}")
add_library(semes_diagnostics STATIC "${REPO}/src/alignment_diagnostics.cpp")
target_include_directories(semes_diagnostics PUBLIC "${REPO}/include" ${OpenCV_INCLUDE_DIRS})
target_link_libraries(semes_diagnostics PUBLIC ${OpenCV_LIBS})
set(J "${REPO}/experiments/hye/v2_Y_T_joint")
add_subdirectory("${J}" joint)
add_subdirectory("${REPO}/experiments/hye/final_pin" final_pin)
add_executable(pin_eval src/main.cpp)
target_include_directories(pin_eval PRIVATE "${J}/include")
target_link_libraries(pin_eval PRIVATE semes_joint semes_final_pin)
if(MSVC)
  target_compile_options(pin_eval PRIVATE /utf-8 /EHsc /O2)
  target_compile_definitions(pin_eval PRIVATE NOMINMAX)
endif()
