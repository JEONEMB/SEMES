cmake_minimum_required(VERSION 3.15)
project(subpixel_real_consistency LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(OpenCV 3.4 REQUIRED)
get_filename_component(REPO "${CMAKE_CURRENT_SOURCE_DIR}/../../.." ABSOLUTE)
set(CONTROL "${REPO}/experiments/hye/00000_final_Y_T control")
# 정합 방법 9개는 1단계 실험 소스를 그대로 쓴다 (같은 설정).
set(STAGE1 "${REPO}/experiments/hye/20261029_122323_부화소정합_합성정답")
add_executable(real_consistency src/main.cpp "${STAGE1}/src/synth.cpp" "${STAGE1}/src/methods.cpp" "${CONTROL}/src/tracker.cpp" "${CONTROL}/src/spline.cpp")
target_include_directories(real_consistency PRIVATE "${STAGE1}/src" "${REPO}/include" "${CONTROL}/include" ${OpenCV_INCLUDE_DIRS})
target_link_libraries(real_consistency PRIVATE ${OpenCV_LIBS} bcrypt)
target_compile_definitions(real_consistency PRIVATE REPO_DIR="${REPO}" EXPERIMENT_DIR="${CMAKE_CURRENT_SOURCE_DIR}")
if(MSVC)
 target_compile_options(real_consistency PRIVATE /utf-8 /EHsc)
 target_compile_definitions(real_consistency PRIVATE NOMINMAX _CRT_SECURE_NO_WARNINGS)
endif()
add_custom_command(TARGET real_consistency POST_BUILD COMMAND ${CMAKE_COMMAND} -E copy_if_different "${OpenCV_DIR}/x64/vc15/bin/opencv_world3415$<$<CONFIG:Debug>:d>.dll" "$<TARGET_FILE_DIR:real_consistency>")
