cmake_minimum_required(VERSION 3.15)
project(qr_subpixel_comparison LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
find_package(OpenCV 3.4 REQUIRED)
set(REPO "${CMAKE_CURRENT_LIST_DIR}/../.." CACHE PATH "QR pipeline repository")
file(GLOB SRC "${REPO}/src/*.cpp")
list(FILTER SRC EXCLUDE REGEX "/main.cpp$")
add_executable(compare compare.cpp ${SRC})
target_include_directories(compare PRIVATE "${REPO}/include")
target_link_libraries(compare PRIVATE ${OpenCV_LIBS})
target_compile_definitions(compare PRIVATE HAS_QR_REFINEMENT_SELECTOR=1)
add_executable(augment augment.cpp)
target_link_libraries(augment PRIVATE ${OpenCV_LIBS})
add_executable(test_refiners test_refiners.cpp)
target_include_directories(test_refiners PRIVATE "${REPO}/include")
target_link_libraries(test_refiners PRIVATE ${OpenCV_LIBS})
add_executable(validate validate.cpp ${SRC})
target_include_directories(validate PRIVATE "${REPO}/include")
target_link_libraries(validate PRIVATE ${OpenCV_LIBS})
add_executable(pipeline_main ${REPO}/src/main.cpp ${SRC})
target_include_directories(pipeline_main PRIVATE "${REPO}/include")
target_link_libraries(pipeline_main PRIVATE ${OpenCV_LIBS})
add_custom_command(TARGET compare POST_BUILD COMMAND ${CMAKE_COMMAND} -E copy_if_different
 "${OpenCV_DIR}/x64/vc15/bin/opencv_world3415$<$<CONFIG:Debug>:d>.dll" "$<TARGET_FILE_DIR:compare>")
