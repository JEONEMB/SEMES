# sdk — 시뮬레이터 추정기 플러그인 만들기
#
#  팀원 : team/<이름>/ 에 자기 파일을 넣고 build_plugin.bat (또는 아래 cmake 명령) → plugins/<이름>/<이름>.dll
#     team/<이름>/include/alignment.hpp      팀 공통 헤더 (Alignment_t alignment(cv::Mat &src);)
#     team/<이름>/include/...               그 밖의 헤더
#     team/<이름>/**/*.cpp                   구현 — 폴더 안 모든 .cpp (main.cpp, sim_hooks.cpp, build/ results/ test(s)/ tool(s)/ 폴더 제외)
#     team/<이름>/sim_hooks.cpp              (선택) 초기화·기준 사진 교체 연결 — 예시 examples/*/sim_hooks.cpp
#     team/<이름>/plugin.cmake               (선택) 추가 설정 : target 이름 = sim_plugin_<이름> (예 : Windows 라이브러리 링크)
#
#  단독 빌드 : cmake -S sdk -B sdk/build -A x64 -DSIM_OPENCV_ROOT="C:/opencv/build" && cmake --build sdk/build --config Release
cmake_minimum_required(VERSION 3.15)
project(SimulatorPlugins LANGUAGES CXX)

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(NOT CMAKE_BUILD_TYPE AND NOT CMAKE_CONFIGURATION_TYPES)
    set(CMAKE_BUILD_TYPE Release)
endif()
if(NOT OpenCV_FOUND)
if(WIN32 AND SIM_OPENCV_ROOT)
        # Windows 배포판 OpenCV 를 직접 지정 (OpenCVConfig.cmake 가 새 Visual Studio 버전을 모를 때도 동작)
        set(OpenCV_INCLUDE_DIRS "${SIM_OPENCV_ROOT}/include")
        set(OpenCV_LIBS optimized "${SIM_OPENCV_ROOT}/x64/vc15/lib/opencv_world3415.lib" debug "${SIM_OPENCV_ROOT}/x64/vc15/lib/opencv_world3415d.lib")
        set(OpenCV_DIR "${SIM_OPENCV_ROOT}")
        set(OpenCV_VERSION "3.4.15 (SIM_OPENCV_ROOT)")
        set(OpenCV_FOUND TRUE)
    else()
        if(NOT OpenCV_DIR AND DEFINED ENV{OPENCV_DIR})
            set(OpenCV_DIR "$ENV{OPENCV_DIR}")
        endif()
        find_package(OpenCV REQUIRED)
    endif()
endif()

set(SIM_SDK_DIR "${CMAKE_CURRENT_SOURCE_DIR}")
set(SIM_PLUGIN_OUT_DIR "${SIM_SDK_DIR}/../plugins" CACHE PATH "만든 플러그인을 둘 폴더 (시뮬레이터 루트의 plugins)")
# 예시 플러그인(ksh_plugin)용 정렬 알고리즘 위치 : 저장소 루트(PJT) 를 먼저, 없으면 예전 ../../Alignment
if(NOT SIM_ALIGNMENT_DIR)
    if(EXISTS "${SIM_SDK_DIR}/../../../src/modules")
        set(SIM_ALIGNMENT_DIR "${SIM_SDK_DIR}/../../..")
    else()
        set(SIM_ALIGNMENT_DIR "${SIM_SDK_DIR}/../../Alignment")
    endif()
endif()
option(SIM_BUILD_EXAMPLE_PLUGIN "예시 플러그인 ksh_plugin 도 만들기 (Alignment 모듈이 있을 때)" ON)

# sim_add_plugin(<이름> SOURCES ... INCLUDES ... [HOOKS 파일])
function(sim_add_plugin name)
    cmake_parse_arguments(A "" "HOOKS" "SOURCES;INCLUDES" ${ARGN})
    if(NOT A_HOOKS)
        set(A_HOOKS "${SIM_SDK_DIR}/sim_hooks_default.cpp")
    endif()
    set(t sim_plugin_${name})
    add_library(${t} SHARED "${SIM_SDK_DIR}/team_alignment_plugin.cpp" "${A_HOOKS}" ${A_SOURCES})
    target_include_directories(${t} PRIVATE "${SIM_SDK_DIR}" ${A_INCLUDES} ${OpenCV_INCLUDE_DIRS})
    target_link_libraries(${t} PRIVATE ${OpenCV_LIBS})
    target_compile_definitions(${t} PRIVATE SIM_PLUGIN_NAME="${name}")
    # 출력 : plugins/<이름>/<이름>.dll (플러그인마다 폴더 — 자산 파일을 옆에 둘 수 있음)
    set(out "${SIM_PLUGIN_OUT_DIR}/${name}")
    set_target_properties(${t} PROPERTIES OUTPUT_NAME "${name}" PREFIX ""
        CXX_VISIBILITY_PRESET hidden VISIBILITY_INLINES_HIDDEN ON
        RUNTIME_OUTPUT_DIRECTORY "${out}" LIBRARY_OUTPUT_DIRECTORY "${out}")
    foreach(cfg Debug Release RelWithDebInfo MinSizeRel)
        string(TOUPPER ${cfg} CFG)
        set_target_properties(${t} PROPERTIES RUNTIME_OUTPUT_DIRECTORY_${CFG} "${out}" LIBRARY_OUTPUT_DIRECTORY_${CFG} "${out}")
    endforeach()
    if(MSVC)
        target_compile_options(${t} PRIVATE /utf-8 /EHsc)
        target_compile_definitions(${t} PRIVATE NOMINMAX _CRT_SECURE_NO_WARNINGS)
    else()
        target_compile_definitions(${t} PRIVATE __error_t_defined=1)
        target_link_options(${t} PRIVATE -Wl,--no-undefined)
    endif()
    message(STATUS "플러그인 ${name} → ${out}")
endfunction()

# ---- 팀원 폴더 : team/<이름>/ 마다 하나
set(SIM_TEAM_DIR "${SIM_SDK_DIR}/team" CACHE PATH "팀원 폴더들이 있는 곳")
file(GLOB SIM_TEAM_DIRS LIST_DIRECTORIES true "${SIM_TEAM_DIR}/*")
foreach(dir ${SIM_TEAM_DIRS})
    if(IS_DIRECTORY "${dir}")
        get_filename_component(name "${dir}" NAME)
        file(GLOB_RECURSE src CONFIGURE_DEPENDS "${dir}/*.cpp" "${dir}/*.cc" "${dir}/*.cxx")
        list(FILTER src EXCLUDE REGEX "/(main|sim_hooks)\\.(cpp|cc|cxx)$")
        list(FILTER src EXCLUDE REGEX "/(build|results|tests?|tools?|examples)/")
        if(NOT src)
            message(WARNING "team/${name} : .cpp 파일이 없어 건너뜀")
            continue()
        endif()
        set(hooks "")
        if(EXISTS "${dir}/sim_hooks.cpp")
            set(hooks "${dir}/sim_hooks.cpp")
        endif()
        sim_add_plugin(${name} SOURCES ${src} INCLUDES "${dir}/include" "${dir}/src" "${dir}" HOOKS ${hooks})
        list(LENGTH src nsrc)
        message(STATUS "  team/${name} : .cpp ${nsrc}개")
        if(EXISTS "${dir}/plugin.cmake")
            include("${dir}/plugin.cmake")
        endif()
    endif()
endforeach()

# ---- 예시 : ksh Alignment 를 플러그인으로 (내장 추정기와 결과가 같아야 함 → 플러그인 경로 검증용)
if(SIM_BUILD_EXAMPLE_PLUGIN AND EXISTS "${SIM_ALIGNMENT_DIR}/src/modules")
    # 저장소 구조 : include/ · src/alignment.cpp · src/modules/*.cpp
    file(GLOB ksh_src "${SIM_ALIGNMENT_DIR}/src/modules/*.cpp")
    sim_add_plugin(ksh_plugin SOURCES "${SIM_ALIGNMENT_DIR}/src/alignment.cpp" ${ksh_src}
        INCLUDES "${SIM_ALIGNMENT_DIR}/include" HOOKS "${SIM_SDK_DIR}/examples/ksh/sim_hooks.cpp")
elseif(SIM_BUILD_EXAMPLE_PLUGIN AND EXISTS "${SIM_ALIGNMENT_DIR}/alignment.cpp")
    # 예전 구조 : alignment.cpp · ksh/src/*.cpp · ksh/include
    file(GLOB ksh_src "${SIM_ALIGNMENT_DIR}/ksh/src/*.cpp")
    sim_add_plugin(ksh_plugin SOURCES "${SIM_ALIGNMENT_DIR}/alignment.cpp" ${ksh_src}
        INCLUDES "${SIM_ALIGNMENT_DIR}" "${SIM_ALIGNMENT_DIR}/ksh/include" HOOKS "${SIM_SDK_DIR}/examples/ksh/sim_hooks.cpp")
endif()
