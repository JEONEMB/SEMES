cmake_minimum_required(VERSION 3.20)
project(qr_common_c VERSION 1.0 LANGUAGES C CXX)
set(CMAKE_C_STANDARD 17)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_WINDOWS_EXPORT_ALL_SYMBOLS ON)
add_library(qr_common_core SHARED src/core.c src/edges.c src/corners.c)
target_include_directories(qr_common_core PUBLIC include PRIVATE calibration)
if(NOT MSVC)
  target_link_libraries(qr_common_core PRIVATE m)
  target_compile_options(qr_common_core PRIVATE -Wall -Wextra -Wpedantic)
else()
  target_compile_options(qr_common_core PRIVATE /W4)
endif()
find_package(OpenCV 3.4 REQUIRED COMPONENTS core imgproc imgcodecs video)
add_library(qr_common_image SHARED src/image_adapter.cpp)
target_include_directories(qr_common_image PRIVATE include calibration ${OpenCV_INCLUDE_DIRS})
target_link_libraries(qr_common_image PRIVATE qr_common_core ${OpenCV_LIBS})
add_executable(qr_common_predict src/main.cpp)
target_include_directories(qr_common_predict PRIVATE include)
target_link_libraries(qr_common_predict PRIVATE qr_common_image qr_common_core)
if(WIN32)
  file(GLOB OPENCV_DLLS "${OpenCV_DIR}/x64/vc*/bin/opencv_world*.dll")
  foreach(DLL ${OPENCV_DLLS})
    add_custom_command(TARGET qr_common_predict POST_BUILD COMMAND ${CMAKE_COMMAND} -E copy_if_different "${DLL}" "$<TARGET_FILE_DIR:qr_common_predict>")
  endforeach()
endif()
